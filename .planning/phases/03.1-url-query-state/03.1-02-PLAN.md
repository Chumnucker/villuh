  <behavior>
    Grammar (D-45): split the param value on `,`; a token containing `:` opens a new predicate (text before the first `:` is the operator, text after is its first value); a token without `:` appends another value to the currently-open predicate.

    - `parse("any:s1,s2")` with allowed `["is_any_of"]` → `[{ operator: "is_any_of", values: ["s1","s2"] }]`
    - `parse("gte:1500,lte:3000")` with allowed `["gte","lte"]` → `[{ operator:"gte", values:["1500"] }, { operator:"lte", values:["3000"] }]`
    - `parse("!any:s1")` with allowed `["is_not_any_of"]` → `[{ operator:"is_not_any_of", values:["s1"] }]`
    - `parse("!empty")` with allowed `["not_empty"]` → `[{ operator:"not_empty", values:[] }]`
    - `parse("bogus:1,gte:2")` with allowed `["gte"]` → only the `gte` predicate; no throw (D-19)
    - `parse("empty")` with allowed `["is_any_of","is_not_any_of"]` (i.e. `idFilterOperators({ nullable:false })`) → `null`; the unsupported operator is dropped and nothing remains
    - `parse("s1")` (a leading continuation value with no open predicate) → `null`, no throw
    - `parse("gte:")` → `null` (a value-bearing operator with no value is dropped)
    - `parse("")` → `null`
    - Bijectivity: `serialize(parse(s))` equals `s` for `"any:s1,s2"`, `"gte:1500,lte:3000"`, `"!empty"`, and `"any:a~~cb"` (a value whose literal text is `a~cb`); `parse(serialize(x))` deep-equals `x` for each of those predicate arrays
    - Escapes (D-45): a value containing `:` serializes as `~c` and a value containing `,` as `~s`; `~` itself doubles to `~~`. `parse(serialize([{operator:"is_any_of",values:[…]}]))` round-trips exactly for the value set `["a:b", "c,d", "e~f", "~c", "~s", "~~", "a~cb"]`. **The last four are the ones that matter** — a value whose literal text is `~c` / `~s` / `~~` is exactly what a sequential three-pass decode gets wrong, and the first three vectors all pass under the broken implementation. Assert each of the seven individually, not just as one combined array, so a failure names the offending value
    - `eq` (D-41): `eq(a, b)` is `true` for two structurally-equal but distinct arrays, `false` when an operator differs, a value differs, a value count differs, or a predicate count differs
    - `defaultValue` is a single frozen module-scope empty array — two separate calls to `parseAsFieldFilters` return parsers whose `defaultValue` is the SAME object reference (D-41 rule 2)
  </behavior>

```ts
import { createParser } from "nuqs"
import type { FieldFilter } from "./types"

/**
 * URL operator alias <-> internal operator (D-19). Only these are accepted;
 * anything else in a hand-edited URL is dropped, never thrown.
 *
 * The internal names must match the vocabulary `data-table-query.ts` already
 * consumes (`INCLUDE_OPERATORS`, `EXCLUDE_OPERATORS`, `getEmptyFilterFlag`,
 * `NUMERIC_OP_ALIASES`) — this is a URL-facing shorthand for that vocabulary,
 * not a second one.
 */
export const URL_OP_TO_INTERNAL = {
  any: "is_any_of",
  "!any": "is_not_any_of",
  empty: "empty",
  "!empty": "not_empty",
  eq: "eq",
  neq: "neq",
  gt: "gt",
  gte: "gte",
  lt: "lt",
  lte: "lte",
} as const

const INTERNAL_TO_URL_OP: Record<string, string> = Object.fromEntries(
  Object.entries(URL_OP_TO_INTERNAL).map(([url, internal]) => [internal, url])
)

/**
 * Operators that carry no values. D-45 note: a valueless operator cannot be
 * distinguished from a value when it trails a value-bearing predicate, so
 * empty / not-empty is EXCLUSIVE per field in the URL — it must be the only
 * predicate on that field. This is already true semantically: `getEmptyFilterFlag`
 * (data-table-query.ts:145-155) collapses conflicting empty chips to newest-wins.
 */
const VALUELESS = new Set<string>(["empty", "not_empty"])

/** Frozen module-scope singleton — a fresh `[]` per render breaks setter stability (D-41). */
const EMPTY_PREDICATES = Object.freeze([]) as unknown as FieldFilter[]

// `~` is not in nuqs's escape set (RESEARCH §11.1), so these survive the encoder
// verbatim. Do NOT use encodeURIComponent — nuqs escapes `%` first, turning
// `%3A` into `%253A` and destroying D-18's hand-editable URLs.
const encodeValue = (v: string) =>
  v.replace(/~/g, "~~").replace(/:/g, "~c").replace(/,/g, "~s")

// ONE left-to-right scan. Do NOT decode with three sequential `.replace` passes
// (`.replace(/~c/g,":").replace(/~s/g,",").replace(/~~/g,"~")`) — that is not
// bijective for any value containing the literal substring `~c` or `~s`:
// `"~c"` encodes to `"~~c"`, and pass 1 then matches the `~c` at index 1 and
// yields `"~:"`. A single scan consumes each escape pair atomically, so `~~`
// is decoded before its second `~` can pair with the following character.
const decodeValue = (v: string) =>
  v.replace(/~(.)/g, (_, c) =>
    c === "c" ? ":" : c === "s" ? "," : c === "~" ? "~" : "~" + c
  )

export function fieldFiltersEq(a: FieldFilter[], b: FieldFilter[]): boolean {
  if (a === b) return true
  if (a.length !== b.length) return false
  return a.every((pa, i) => {
    const pb = b[i]!
    return (
      pa.operator === pb.operator &&
      pa.values.length === pb.values.length &&
      pa.values.every((v, j) => v === pb.values[j])
    )
  })
}

/**
 * One filter field's predicates, e.g. `gte:1500,lte:3000` or `any:s1,s2` (D-18).
 *
 * @param fieldKey          only used for the dev-time serializer warning
 * @param allowedOperators  the field's declared internal operator list (D-19);
 *                          anything outside it is dropped rather than thrown
 */
export function parseAsFieldFilters(
  fieldKey: string,
  allowedOperators: readonly string[]
) {
  const allowed = new Set(allowedOperators)

  return createParser<FieldFilter[]>({
    parse(query) {
      const out: FieldFilter[] = []
      for (const token of query.split(",")) {
        const colon = token.indexOf(":")
        if (colon === -1) {
          // Continuation value for the open predicate; a stray leading one is ignored.
          out[out.length - 1]?.values.push(decodeValue(token))
          continue
        }
        const internal =
          URL_OP_TO_INTERNAL[token.slice(0, colon) as keyof typeof URL_OP_TO_INTERNAL]
        if (!internal || !allowed.has(internal)) continue // drop, never throw (D-19)
        const raw = token.slice(colon + 1)
        out.push({
          operator: internal,
          values: raw === "" ? [] : [decodeValue(raw)],
        })
      }
      // A value-bearing operator that ended up with no values is meaningless.
      const cleaned = out.filter(
        (p) => VALUELESS.has(p.operator) || p.values.length > 0
      )
      return cleaned.length > 0 ? cleaned : null
    },

    serialize(predicates) {
      if (process.env.NODE_ENV !== "production") {
        for (const p of predicates) {
          for (const v of p.values) {
            if (v.includes(":") || v.includes(",") || v.includes("~")) {
              console.warn(
                `[query-state] Filter value for "${fieldKey}" contains a grammar delimiter and will be escaped: ${JSON.stringify(v)}. ` +
                  `All current filter values are UUIDs or numbers; if this is a new option source, confirm the escaped URL still reads well. See D-45.`
              )
            }
          }
        }
      }
      return predicates
        .map((p) =>
          VALUELESS.has(p.operator)
            ? INTERNAL_TO_URL_OP[p.operator]!
            : `${INTERNAL_TO_URL_OP[p.operator]}:${p.values.map(encodeValue).join(",")}`
        )
        .join(",")
    },

    // NOT optional. `clearOnDefault` compares against defaultValue with `eq`
    // (nuqs dist/index.js:576), AND the keyMap-stability heuristic calls
    // `parser.eq?.(previousDefault, currentDefault)` (dist/index.js:494).
    // Omitting it costs both a clean URL (D-20) and a stable setter (D-41).
    eq: fieldFiltersEq,
  }).withDefault(EMPTY_PREDICATES)
}
```

Bare `empty` / `!empty` (no colon) is intentionally NOT parsed as an operator — under D-45's grammar a token without `:` is a continuation value. The canonical serialized form for a valueless predicate is the bare word, which only ever appears as the *first* token of a field that has no other predicates. Handle it: before the token loop, if the whole `query` is exactly a valueless URL alias (`"empty"` or `"!empty"`) and that internal operator is allowed, return that single predicate directly. Add a test for `parse("empty")` with allowed `["empty"]` → `[{ operator:"empty", values:[] }]`.

For the test file, use `isParserBijective` from `nuqs/testing` for the round-trip cases in addition to the explicit assertions. Import as `import { isParserBijective } from "nuqs/testing"`.

Its case list must include **at least one escape-bearing value**, not only `"any:s1,s2"` / `"gte:1500,lte:3000"` / `"!empty"`. Those three contain no `~`, so they round-trip cleanly under a broken sequential decode and the suite goes green over the bug. Add `"any:a~~cb"` (serialized form of the single value `a~cb`) to the list.

D-24 note for the reader and for later plans: the parser deliberately returns `{ operator, values }` with **no `id`**. The deterministic chip id is minted by `useDataTableQueryState` (plan 03.1-03) as `` `${fieldKey}:${operator}:${index}` `` so the parser's `eq` never compares an id it does not own.

import {
  createParser,
  parseAsIndex,
  parseAsNumberLiteral,
  parseAsString,
} from "nuqs"
import { DATA_TABLE_DEFAULT_PAGE_SIZE, DATA_TABLE_PAGE_SIZES } from "../data-table"
import type { DataTableSort } from "../data-table-query"

/**
 * `sort=-baseRent` <-> `{ id: "baseRent", desc: true }` (D-20).
 * Validated against the table's declared sortable column ids — an unknown id
 * yields null, which nuqs resolves to the default (no sort).
 */
export function parseAsSort(sortableIds: readonly string[]) {
  const allowed = new Set(sortableIds)
  return createParser<DataTableSort>({
    parse(query) {
      const desc = query.startsWith("-")
      const id = desc ? query.slice(1) : query
      return allowed.has(id) ? { id, desc } : null
    },
    serialize: (s) => (s.desc ? `-${s.id}` : s.id),
    // Non-primitive value -> `eq` is required for clearOnDefault and for
    // keyMap stability (D-41). Note this parser has NO default, so its
    // `defaultValue` is `undefined` and stable by identity; `eq` still matters

}
