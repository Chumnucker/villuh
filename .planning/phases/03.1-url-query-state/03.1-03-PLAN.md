```tsx
"use client"

import * as React from "react"
import { PlatformIcon } from "../icon"
import type { Filter, FilterFieldConfig } from "../filters"
import type {
  FieldFilter,
  FilterOptionItem,
  TableConfig,
  TableDefinition,
  TableFieldDef,
  TableFieldUi,
} from "./types"
import { registerQueryStateKeys } from "./key-registry"
import {
  pageParser,
  parseAsFieldFilters,
  parseAsSort,
  searchParser,
  sizeParser,
} from "./table-params"

const EMPTY_OPTIONS: FilterOptionItem[] = Object.freeze([]) as FilterOptionItem[]

/** Id-list (multiselect) filter field. Pair with `idFilterOperators()`. */
export function ids(
  def: TableFieldUi & {
    operators: TableFieldDef["operators"]
    server: Extract<TableFieldDef["server"], { kind: "ids" }>
    useOptions?: TableFieldDef["useOptions"]
  }
): TableFieldDef {
  return { ...def, kind: "ids" }
}

/** Number filter field. Pair with `numericFilterOperators()`. */
export function numeric(
  def: TableFieldUi & {
    operators: TableFieldDef["operators"]
    server: Extract<TableFieldDef["server"], { kind: "numeric" }>
    useOptions?: TableFieldDef["useOptions"]
  }
): TableFieldDef {
  return { ...def, kind: "numeric" }
}

/**
 * One config object per table (D-21).
 *
 * `name` is a definition-unique identifier — `"units"`, `"leases"` — and is
 * INDEPENDENT of `namespace`. It is the owner string this definition registers
 * its URL keys under. Do not derive it from `namespace`; see the note on the
 * `registerQueryStateKeys` call below for why that breaks D-29.
 *
 * `namespace` is `null` for the normal case — one table per page, so keys read
 * best unprefixed (`?q=main&status=any:s1&sort=-baseRent&page=2&size=50`).
 * Pass a namespace only when a single page hosts two tables; keys then become
 * `${namespace}:${key}` (D-29 — table prefixes use `:`, overlay namespaces use `.`,
 * so the two schemes can never converge).
 *
 * Everything below runs ONCE, at module evaluation. The parser map and the
 * urlKeys object are frozen and never rebuilt, which is requirement 2 of D-41 —
 * a keyMap rebuilt per render destabilizes nuqs's `stableKeyMap` and permanently
 * re-arms DataTableToolbar's 300 ms search debounce.
 */
export function defineTable<const F extends Record<string, TableFieldDef>>(
  name: string,
  namespace: string | null,
  config: TableConfig<F>
): TableDefinition<F> {
  const fieldKeys = Object.keys(config.fields) as (keyof F & string)[]
  const entries = fieldKeys.map((key) => [key, config.fields[key]!] as const)
  const prefix = namespace ? `${namespace}:` : ""

  const parsers = Object.freeze({
    q: searchParser,
    sort: parseAsSort(config.sortable),
    page: pageParser,
    size: sizeParser,
    ...Object.fromEntries(
      entries.map(([key, def]) => [
        key,
        parseAsFieldFilters(key, def.operators.map((op) => op.value)),
      ])
    ),
  })

  const urlKeys = Object.freeze(
    Object.fromEntries(
      Object.keys(parsers).map((key) => [key, `${prefix}${key}`])
    )
  ) as Readonly<Record<string, string>>

  // Owner is `name`, NOT `namespace ?? "table"`. D-29's stated trigger is a
  // second table on one page — which is a second UNPREFIXED table. Under a
  // namespace-derived owner both would register `q`/`sort`/`page`/`size` under
  // the literal owner "table", key-registry.ts would see same-owner
  // re-registration, treat it as HMR, and return silently — the assert would be
  // inert for the exact scenario it exists to catch. Keyed on `name`, the second
  // definition registers a different owner against the same keys and throws.
  registerQueryStateKeys(name, Object.values(urlKeys), "table")

  /**
   * D-22 binder. The config is a frozen module-scope object whose entry order
   * never changes, so this loop unrolls to a fixed hook sequence. Fields are
   * NEVER conditionally skipped — a hidden field's `useOptions` still runs and
   * receives `hidden: true` so it can gate its own tRPC query; only the RESULT
   * is filtered. Skipping the iteration instead is Pitfall 2: "Rendered fewer
   * hooks than during the previous render" when navigating between /units and a
   * property Units tab.
   */
  function useFilterFields(options?: { hide?: readonly string[] }): FilterFieldConfig[] {
    const hide = options?.hide
    const resolved: FilterFieldConfig[] = []

    for (const [key, def] of entries) {
      const hidden = hide?.includes(key) ?? false
      // eslint-disable-next-line react-hooks/rules-of-hooks -- static, frozen, module-scope config; entry order never changes and no field is ever conditionally skipped, so the hook sequence is fixed. See D-22 / RESEARCH §6.
      const opts = def.useOptions?.(hidden) ?? EMPTY_OPTIONS
      if (hidden) continue
      const { icon, useOptions: _useOptions, kind: _kind, ...ui } = def
      resolved.push({
        ...ui,
        key,
        type: def.kind === "ids" ? "multiselect" : "text",
        icon: icon ? <PlatformIcon icon={icon} size={16} /> : undefined,
        options: def.kind === "ids" ? opts : undefined,
        operators: def.operators,
        server: def.server,
      })
    }

    return resolved
  }

  /** Inverse of the parse side: `DataTableQuery.filters` -> one param value per field. */
  function toFieldParams(filters: Filter[]): Record<string, FieldFilter[] | null> {
    const out: Record<string, FieldFilter[] | null> = {}
    for (const key of fieldKeys) {
      const predicates = filters
        .filter((f) => f.field === key)
        .map((f) => ({ operator: f.operator, values: f.values.map(String) }))
      // `null` clears the param; clearOnDefault would also clear it against the
      // frozen empty default, but null is explicit and one fewer comparison.
      out[key] = predicates.length > 0 ? predicates : null
    }
    return out
  }

  return { name, namespace, fieldKeys, parsers, urlKeys, useFilterFields, toFieldParams }
}
```

```ts
"use client"

import * as React from "react"
import { useQueryStates } from "nuqs"
import { DATA_TABLE_DEFAULT_PAGE_SIZE } from "../data-table"
import {
  buildDataTableServerInput,
  type DataTableQuery,
} from "../data-table-query"
import type { Filter, FilterFieldConfig } from "../filters"
import type { FieldFilter, TableDefinition, TableFieldDef } from "./types"

/**
 * D-26: table state is `replace`. Nobody should need twelve back presses to
 * escape a table they were filtering.
 *
 * D-42 caveat: nuqs's throttle queue is a global singleton that merges every
 * update inside a ~150 ms window into one history write and applies `push` if
 * ANY contributing update asked for it. A table update coalesced with an overlay
 * open therefore inherits `push`. That is acceptable — one history entry
 * capturing both the overlay and the table state is exactly the shareable-link
 * goal — but it means D-26 holds only for table changes with no overlay
 * interaction. Do not fire an overlay open and a table update from one handler.
 */
const TABLE_OPTIONS = { history: "replace", scroll: false } as const

/**
 * URL-backed twin of `useDataTableQuery` (D-28). Identical return signature plus
 * `filterFields`, so a call site swaps one hook and nothing else — `DataTable`,
 * `DataGrid`, the column defs, and `DataTableToolbar` are untouched.
 */
export function useDataTableQueryState<TInput = Record<string, unknown>>(
  table: TableDefinition<Record<string, TableFieldDef>>,
  options?: { hide?: readonly string[] }
) {
  // `table.parsers` / `table.urlKeys` are frozen module-scope objects built by
  // defineTable — requirement 2 of D-41. Never build a keyMap in this body.
  const [values, setParams] = useQueryStates(
    table.parsers as Parameters<typeof useQueryStates>[0],
    { ...TABLE_OPTIONS, urlKeys: table.urlKeys }
  )

  const filterFields: FilterFieldConfig[] = table.useFilterFields(options)

  const query = React.useMemo<DataTableQuery>(() => {
    const filters: Filter[] = []
    for (const key of table.fieldKeys) {
      const predicates = (values as Record<string, unknown>)[key] as FieldFilter[]
      predicates.forEach((p, index) => {
        filters.push({
          // D-24: deterministic. Two parses of the same URL yield the same ids,
          // so React keys and chip identity are stable across renders.
          id: `${key}:${p.operator}:${index}`,
          field: key,
          operator: p.operator,
          values: p.values,
        })
      })
    }
    return {
      search: (values as Record<string, unknown>).q as string,
      filters,
      sorting: ((values as Record<string, unknown>).sort as DataTableQuery["sorting"]) ?? undefined,
      pageIndex: (values as Record<string, unknown>).page as number,
      pageSize: (values as Record<string, unknown>).size as number,
    }
  }, [values, table])

  // D-25 / D-41 requirement 1. `setParams` is nuqs's own useCallback-wrapped
  // setter; its only unstable dependency would be `stableKeyMap`, which cannot
  // churn here because the parser map is frozen at module scope and every
  // non-primitive default declares `eq`. If this identity ever changes per
  // render, DataTableToolbar re-arms its 300 ms timer forever and the search
  // box never commits (Pitfall 1).
  const setQuery = React.useCallback(
    (next: DataTableQuery) => {
      // ONE setter call -> ONE history entry (RESEARCH §4). Do not split this.
      void setParams({
        q: next.search || null,
        sort: next.sorting ?? null,
        page: next.pageIndex ?? 0,
        size: next.pageSize ?? DATA_TABLE_DEFAULT_PAGE_SIZE,
        ...table.toFieldParams(next.filters),
      } as never)
    },
    [setParams, table]
  )

  // D-30: the whole data-table-query.ts mapping layer is kept as-is.
  const serverInput = React.useMemo(
    () => buildDataTableServerInput(query, filterFields) as TInput,
    [query, filterFields]
  )

  return { query, setQuery, serverInput, filterFields }
}
```

Two details that decide whether this works:

1. **`table` in the dependency arrays is safe** — `defineTable` returns one object per module, created at module evaluation, so its identity never changes. It is listed for `exhaustive-deps` honesty, not because it can churn.
2. **`query.filters` reference identity.** `DataTable.handleQueryChange` (`data-table.tsx:192-196`) compares `next.filters !== query.filters` **by reference** to decide whether to snap back to page 1. That keeps working because nuqs preserves parsed-value identity while the query string is unchanged (RESEARCH § 5.3: `parseMap` returns the cached state value when `compareQuery` matches), and the `useMemo` above is keyed on the `values` object. Verify this during the units migration (plan 03.1-05). If it ever misfires, the fix is a content comparison **inside this hook** — D-28 forbids changing `DataTable`.

Do not add `clearOnDefault` — `true` is the nuqs v2 default and D-20's clean URL for an untouched table comes for free, provided every parser has a default and a correct `eq` (both handled in plan 03.1-02).

Do not add `limitUrlUpdates` / `throttleMs`. The adapter's `rateLimitFactor: 3` already gives a ~150 ms effective window, and the search debounce stays in `DataTableToolbar` (D-27) — nuqs only ever sees committed search values.
