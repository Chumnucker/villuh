```json
  "./app": "./src/app.ts",
  "./auth": "./src/auth.ts"
}
```

```tsx
"use client"

import { Button } from "@/modules/common/ui/button"
import { PlatformIcon } from "@/modules/common/ui/icon"
import { authClient } from "@/lib/auth-client"
import { toast } from "sonner"
import { useState } from "react"

export function GoogleButton({ callbackURL = "/" }: { callbackURL?: string }) {
  const [loading, setLoading] = useState(false)
  return (
    <Button
      type="button"
      variant="outline"
      size="lg"
      className="w-full"
      disabled={loading}
      onClick={async () => {
        setLoading(true)
        try {
          await authClient.signIn.social({ provider: "google", callbackURL })

          toast.error("Something went wrong. Please try again.")
          setLoading(false)
        }
      }}
    >
      <PlatformIcon icon="IconGoogle" size={18} strokeWidth={2} />
      Continue with Google
    </Button>
  )
}
```

```tsx
import { useState } from "react"
import { Button, LinkButton } from "@/modules/common/ui/button"
import { Input, PasswordInput } from "@/modules/common/ui/input"
import { Field, FieldLabel } from "@/modules/common/ui/field"
import { authClient } from "@/lib/auth-client"

  const [name, setName] = useState("")
  const [email, setEmail] = useState("")
  const [password, setPassword] = useState("")
  const [submitted, setSubmitted] = useState(false)
  const [loading, setLoading] = useState(false)

  if (submitted) {
    return (
      <AuthCard
        title="Check your email"
        description={`We sent a verification link to ${email}. Click it to activate your account.`}
      >
        <p className="text-sm text-muted-foreground">

        </p>
      </AuthCard>
    )

  return (
    <AuthCard
      title="Create your account"
      description="30 days free. No credit card required."
      footer={
        <p className="text-sm text-muted-foreground">
          Already have an account?{" "}
          <LinkButton href="/login" variant="link">Sign in</LinkButton>
        </p>

      <form
        className="flex flex-col gap-4"

            if (error.message?.toLowerCase().includes("already")) {

            return

        <Field>
          <FieldLabel>Email</FieldLabel>
          <Input type="email" value={email} onChange={(e) => setEmail(e.target.value)} required autoComplete="email" />
        </Field>
        <Field>
          <FieldLabel>Password</FieldLabel>
          <PasswordInput value={password} onChange={(e) => setPassword(e.target.value)} required autoComplete="new-password" minLength={8} />
        </Field>
        <Button type="submit" size="lg" className="w-full" disabled={loading}>
          Create account
        </Button>
        <div className="relative my-2">
          <div className="absolute inset-0 flex items-center"><span className="w-full border-t" /></div>
          <div className="relative flex justify-center text-xs uppercase">
            <span className="bg-card px-2 text-muted-foreground">or</span>
          </div>
        </div>
        <GoogleButton callbackURL="/onboarding" />
```

**Step 2.6 — Create `packages/web/modules/auth/components/reset-password-form.tsx`:**

Reads token from `useSearchParams`. Two PasswordInput fields (new + confirm). Mismatch sets `aria-invalid` on confirm field. On success, replace card with "Password updated" + LinkButton → /login. Invalid token: detect `error.message` containing "expired" or "invalid" and show "Link expired" card with `LinkButton` "Request a new one" → `/forgot-password`.

```tsx
"use client"

import { useState } from "react"
import { useSearchParams } from "next/navigation"
import { Button, LinkButton } from "@/modules/common/ui/button"
import { PasswordInput } from "@/modules/common/ui/input"
import { Field, FieldLabel, FieldError } from "@/modules/common/ui/field"
import { authClient } from "@/lib/auth-client"
import { toast } from "sonner"
import { AuthCard } from "./auth-card"

export function ResetPasswordForm() {
  const params = useSearchParams()
  const token = params.get("token") ?? ""
  const [newPassword, setNewPassword] = useState("")
  const [confirm, setConfirm] = useState("")
  const [mismatch, setMismatch] = useState(false)
  const [done, setDone] = useState(false)
  const [expired, setExpired] = useState(false)
  const [loading, setLoading] = useState(false)

  if (done) {
    return (
      <AuthCard title="Password updated" description="You can now sign in with your new password.">
        <LinkButton href="/login" variant="default" className="w-full">Sign in</LinkButton>
      </AuthCard>
    )
  }

  if (expired) {
    return (
      <AuthCard
        title="Link expired"
        description="This reset link has expired or already been used."
      >
        <LinkButton href="/forgot-password" variant="link">Request a new one</LinkButton>
      </AuthCard>
    )
  }

  return (
    <AuthCard title="Set a new password">
      <form
        className="flex flex-col gap-4"
        onSubmit={async (e) => {
          e.preventDefault()
          if (newPassword !== confirm) {
            setMismatch(true)
            return
          }
          setMismatch(false)
          setLoading(true)
          const { error } = await authClient.resetPassword({ newPassword, token })
          setLoading(false)
          if (error) {
            const msg = (error.message ?? "").toLowerCase()
            if (msg.includes("expired") || msg.includes("invalid")) {
              setExpired(true)
              return
            }
            toast.error("Something went wrong. Please try again.")
            return
          }
          setDone(true)
        }}
      >
        <Field>
          <FieldLabel>New password</FieldLabel>
          <PasswordInput value={newPassword} onChange={(e) => setNewPassword(e.target.value)} required minLength={8} autoComplete="new-password" />
        </Field>
        <Field>
          <FieldLabel>Confirm password</FieldLabel>
          <PasswordInput value={confirm} onChange={(e) => setConfirm(e.target.value)} required minLength={8} autoComplete="new-password" aria-invalid={mismatch || undefined} />
          {mismatch && <FieldError>Passwords do not match</FieldError>}
        </Field>
        <Button type="submit" size="lg" className="w-full" disabled={loading}>
          Set new password
        </Button>
      </form>
    </AuthCard>
  )
}
```
