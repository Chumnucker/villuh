**Step 1.1 — Create `packages/server/src/middleware/check-plan-limit.ts`:**

Helper function (NOT middleware — called at the top of route handlers per D-41).

```typescript
import type { Context } from 'hono'
import { HTTPException } from 'hono/http-exception'
import { db } from '@villuh/db'
import { organization } from '@villuh/db/schema/auth'
import { planLimits } from '@villuh/db/schema/plan-limits'
import { and, eq, sql } from 'drizzle-orm'
import type { AuthVariables } from './auth.js'

export type Resource = 'units' | 'properties' | 'members'

function nextTier(tier: string): 'growth' | 'pro' {
  if (tier === 'starter') return 'growth'
  return 'pro'
}

/**
 * Throw 402 PLAN_LIMIT (D-42) if the org is at or over its limit for the given resource.
 * Resolves tier from org.subscription_tier (denormalized per D-23).
 * Treats 'trial' tier as 'pro' for limit lookup (D-19 — full pro access during trial).
 * `currentCount` is the count BEFORE the proposed creation (caller passes pre-create count).
 */
export async function checkPlanLimit(
  c: Context<{ Variables: AuthVariables }>,
  resource: Resource,
  currentCount: number
): Promise<void> {
  const session = c.get('session')
  const orgId = (session as { activeOrganizationId?: string } | null)?.activeOrganizationId
  if (!orgId) throw new HTTPException(401, { message: 'No active organization' })

  const orgRow = await db.select().from(organization).where(eq(organization.id, orgId)).limit(1)
  if (orgRow.length === 0) throw new HTTPException(404, { message: 'Organization not found' })
  const tier = (orgRow[0] as { subscriptionTier?: string }).subscriptionTier ?? 'trial'
  const lookupTier = tier === 'trial' ? 'pro' : tier

  const limitRow = await db
    .select()
    .from(planLimits)
    .where(and(eq(planLimits.tier, lookupTier), eq(planLimits.resource, resource)))
    .limit(1)
  if (limitRow.length === 0) {
    // No row found — fail closed (deny) rather than silently allowing
    throw new HTTPException(500, { message: `plan_limits row missing for ${lookupTier} × ${resource}` })
  }
  const limit = limitRow[0].limit
  if (limit === -1) return                                                  // -1 = unlimited

  if (currentCount >= limit) {
    throw new HTTPException(402, {
      message: JSON.stringify({
        code: 'PLAN_LIMIT',
        resource,
        limit,
        current: currentCount,
        upgrade_to: nextTier(tier === 'trial' ? 'pro' : tier),
      }),
    })
  }
}
```

**Step 1.2 — Create `packages/server/src/middleware/subscription-gate.ts`:**

```typescript
import { createMiddleware } from 'hono/factory'
import { db } from '@villuh/db'
import { organization } from '@villuh/db/schema/auth'
import { eq } from 'drizzle-orm'
import type { AuthVariables } from './auth.js'

/**
 * Block all writes when org.subscription_status === 'expired' (D-25, D-43).
 * 402 SUBSCRIPTION_REQUIRED. Read-only gate — apply on POST/PUT/DELETE/PATCH routes.
 *
 * Phase 2 hard-blocks ONLY on 'expired' (per D-19/D-33/D-39 reconciliation):
 * - 'trial_pending' (default for new users): NOT blocked — user is in pre-trial-start onboarding flow
 *   and must be able to call /api/wizard/* + /api/billing/trial-start.
 * - 'trialing'      : NOT blocked — active trial.
 * - 'active'        : NOT blocked — paid subscription.
 * - 'past_due'      : NOT blocked in Phase 2 (Deferred Ideas: grace period).
 * - 'payment_failed': NOT blocked in Phase 2.
 * - 'expired'       : BLOCKED with 402 SUBSCRIPTION_REQUIRED.
 * - 'canceled'      : NOT blocked in Phase 2 (no UI surface for cancel yet).
 *
 * The /api/wizard/* and /api/billing/trial-start routes are reachable for trial_pending users
 * because this gate only blocks 'expired'.
 */
export const subscriptionExpiredGate = createMiddleware<{ Variables: AuthVariables }>(async (c, next) => {
  const user = c.get('user')
  const session = c.get('session')
  if (!user || !session) { await next(); return }            // unauth handled elsewhere
  const orgId = (session as { activeOrganizationId?: string }).activeOrganizationId
  if (!orgId) { await next(); return }

  const orgRow = await db.select().from(organization).where(eq(organization.id, orgId)).limit(1)
  const status = (orgRow[0] as { subscriptionStatus?: string } | undefined)?.subscriptionStatus
  // Only block 'expired' — D-19/D-33/D-39 reconciliation: trial_pending users must reach the wizard.
  if (status === 'expired') {
    return c.json({ code: 'SUBSCRIPTION_REQUIRED', message: 'Subscription expired' }, 402)
  }
  await next()
})
```

```typescript
import { Hono } from 'hono'
import { z } from 'zod'
import { zValidator } from '@hono/zod-validator'
import { authMiddleware, type AuthVariables } from '../middleware/auth.js'
import { requireTenant } from '../middleware/require-tenant.js'
import { requirePermission } from '../middleware/require-permission.js'
import { onboardingState } from '@villuh/db/schema/onboarding-state'
import { organization } from '@villuh/db/schema/auth'
import { withTenant } from '../lib/tenant.js'
import { eq } from 'drizzle-orm'
import { checkPlanLimit } from '../middleware/check-plan-limit.js'

const propertySchema = z.object({
  name: z.string().min(1),
  address: z.string().min(1),
  city: z.string().min(1),
  state: z.string().min(1),
  postal_code: z.string().min(1),
})
const unitSchema = z.object({
  label: z.string().min(1),
  base_rent: z.number().nonnegative(),
})
const tenantSchema = z.object({
  name: z.string().min(1),
  email: z.string().email(),
})

export const wizard = new Hono<{ Variables: AuthVariables }>()
  // GET /api/wizard/state — derive current step from DB (D-35)
  .get('/state', authMiddleware, async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    return withTenant(landlordId, userId, async (tx) => {
      const stateRows = await tx.select().from(onboardingState).where(eq(onboardingState.landlordId, landlordId)).limit(1)
      const orgRows = await tx.select().from(organization).where(eq(organization.id, landlordId)).limit(1)
      const state = stateRows[0]
      const org = orgRows[0]
      const hasProperty = !!(state?.propertyData)
      const hasUnit = !!(state?.unitData)
      const hasTenant = !!(state?.tenantData)
      const setupCompleted = (org as { setupCompleted?: boolean })?.setupCompleted ?? false
      let currentStep: 1 | 2 | 3 | 4 | 'done' = 1
      if (setupCompleted) currentStep = 'done'
      else if (hasTenant) currentStep = 4
      else if (hasUnit) currentStep = 3
      else if (hasProperty) currentStep = 2
      return c.json({
        currentStep,
        propertyData: state?.propertyData ?? null,
        unitData: state?.unitData ?? null,
        tenantData: state?.tenantData ?? null,
        setupCompleted,
      })
    })
  })
  // POST /api/wizard/property — Step 1
  .post('/property', authMiddleware, requirePermission('admin'), zValidator('json', propertySchema), async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    const data = c.req.valid('json')
    // Plan limit on properties — for Phase 2 the count is "1 if property data already set, else 0"
    return withTenant(landlordId, userId, async (tx) => {
      const existing = await tx.select().from(onboardingState).where(eq(onboardingState.landlordId, landlordId)).limit(1)
      const hadProperty = !!(existing[0]?.propertyData)
      // checkPlanLimit only matters when CREATING new — not when updating. Wizard allows back-edit (D-34).
      if (!hadProperty) await checkPlanLimit(c, 'properties', 0)              // 0 because no property exists yet
      await tx.insert(onboardingState).values({ landlordId, propertyData: data })
        .onConflictDoUpdate({ target: onboardingState.landlordId, set: { propertyData: data, updatedAt: new Date() } })
      return c.json({ ok: true })
    })
  })
  // POST /api/wizard/unit — Step 2
  .post('/unit', authMiddleware, requirePermission('admin'), zValidator('json', unitSchema), async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    const data = c.req.valid('json')
    return withTenant(landlordId, userId, async (tx) => {
      const existing = await tx.select().from(onboardingState).where(eq(onboardingState.landlordId, landlordId)).limit(1)
      const hadUnit = !!(existing[0]?.unitData)
      if (!hadUnit) await checkPlanLimit(c, 'units', 0)
      // Step 1 must be complete before step 2
      if (!existing[0]?.propertyData) {
        return c.json({ error: 'Property must be created first' }, 400)
      }
      await tx.insert(onboardingState).values({ landlordId, unitData: data })
        .onConflictDoUpdate({ target: onboardingState.landlordId, set: { unitData: data, updatedAt: new Date() } })
      return c.json({ ok: true })
    })
  })
  // POST /api/wizard/tenant — Step 3
  .post('/tenant', authMiddleware, requirePermission('admin'), zValidator('json', tenantSchema), async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    const data = c.req.valid('json')
    return withTenant(landlordId, userId, async (tx) => {
      const existing = await tx.select().from(onboardingState).where(eq(onboardingState.landlordId, landlordId)).limit(1)
      if (!existing[0]?.unitData) {
        return c.json({ error: 'Unit must be created first' }, 400)
      }
      await tx.insert(onboardingState).values({ landlordId, tenantData: data })
        .onConflictDoUpdate({ target: onboardingState.landlordId, set: { tenantData: data, updatedAt: new Date() } })
      return c.json({ ok: true })
    })
  })
```

**Step 2.1 — Create `packages/web/middleware.ts`:**

Per RESEARCH.md Pattern 6 + UI-SPEC §"Auth layout" exclusion list. Uses Next.js Node.js runtime (Next 15.2+) so it can directly call `auth.api.getSession` and run a Drizzle query.

CRITICAL: Imports `auth` from `@/lib/auth-server` (standalone `betterAuth()` instance per Plan 02-03 Task 1). The standalone instance does NOT have the org plugin, so `auth.api.getFullOrganization` is NOT available. Read `setup_completed` directly from the `organization` table via Drizzle (the `@villuh/db` schema is shared between the Hono server and the Next.js middleware).

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { auth } from '@/lib/auth-server'
import { drizzle } from 'drizzle-orm/postgres-js'
import postgres from 'postgres'
import { organization } from '@villuh/db/schema/auth'
import { eq } from 'drizzle-orm'

export const config = {
  runtime: 'nodejs',
  matcher: ['/((?!api|_next/static|_next/image|favicon.ico|fonts).*)'],
}

// Singleton DB client for middleware setup_completed lookup. Uses villuh_app role per CLAUDE.md.
// RLS does NOT apply here because middleware does not call withTenant — but we only read
// the org row matching session.activeOrganizationId, and the session itself is the gate
// (signed cookie → user → membership → activeOrganizationId).
const middlewareClient = postgres(process.env.DATABASE_APP_URL!, { max: 2 })
const middlewareDb = drizzle({ client: middlewareClient })

const PUBLIC_PATHS = ['/login', '/register', '/forgot-password', '/reset-password', '/accept-invitation']

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl
  const isPublic = PUBLIC_PATHS.some((p) => pathname === p || pathname.startsWith(p + '/'))

  // Allow static + public assets
  if (pathname.startsWith('/_next/') || pathname === '/favicon.ico') {
    return NextResponse.next()
  }

  // Resolve session via the standalone Better Auth instance (Node.js runtime allows direct DB access)
  const session = await auth.api.getSession({ headers: request.headers })

  // Public route + already authenticated → redirect to home
  if (isPublic && session?.user) {
    return NextResponse.redirect(new URL('/', request.url))
  }
  if (isPublic) return NextResponse.next()

  // Protected route + not authenticated → /login
  if (!session?.user) {
    return NextResponse.redirect(new URL('/login', request.url))
  }

  // Authenticated but no active org (race condition during signup) → allow through; org auto-create hook will fix
  const orgId = (session.session as { activeOrganizationId?: string } | undefined)?.activeOrganizationId
  if (!orgId) return NextResponse.next()

  // Direct Drizzle query to read setup_completed and subscription_status from the org row.
  // Pitfall 7 (RESEARCH.md): this adds 1 DB call per request — acceptable for Phase 2.
  // Optimize later via cached header from Hono /api/me if perf becomes a concern.
  let setupCompleted = false
  let subscriptionStatus: string = 'trial_pending'
  try {
    const orgRows = await middlewareDb
      .select()
      .from(organization)
      .where(eq(organization.id, orgId))
      .limit(1)
    const org = orgRows[0] as { setupCompleted?: boolean; setup_completed?: boolean; subscriptionStatus?: string; subscription_status?: string } | undefined
    setupCompleted = org?.setupCompleted ?? org?.setup_completed ?? false
    subscriptionStatus = org?.subscriptionStatus ?? org?.subscription_status ?? 'trial_pending'
  } catch {
    // If lookup fails, assume not yet set up
    setupCompleted = false
  }

  // D-19/D-33/D-39: trial_pending users (just signed up, haven't hit wizard step 4) MUST be allowed into /onboarding.
  // Only setup_completed=true is the "done" state. Subscription gate (server-side) blocks 'expired' only.
  if (!pathname.startsWith('/onboarding') && !setupCompleted) {
    return NextResponse.redirect(new URL('/onboarding', request.url))
  }
  if (pathname.startsWith('/onboarding') && setupCompleted) {
    return NextResponse.redirect(new URL('/', request.url))
  }
  return NextResponse.next()
}
```
