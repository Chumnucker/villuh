**Step 1.1 — Create `packages/server/src/routes/billing/stripe-connect.ts`:**

```typescript
import { Hono } from 'hono'
import { authMiddleware, type AuthVariables } from '../../middleware/auth.js'
import { requireTenant } from '../../middleware/require-tenant.js'
import { requirePermission } from '../../middleware/require-permission.js'
import { stripeClient } from '../../auth.js'
import { db } from '@villuh/db'
import { organization } from '@villuh/db/schema/auth'
import { withTenant } from '../../lib/tenant.js'
import { eq } from 'drizzle-orm'
import { HTTPException } from 'hono/http-exception'

export const stripeConnect = new Hono<{ Variables: AuthVariables }>()
  // GET /callback — Stripe redirects here after the user authorizes the OAuth app.
  // We rely on cookie-auth (the user is signed in to Villuh in the same browser).
  .get('/callback', authMiddleware, async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    const code = c.req.query('code')
    const state = c.req.query('state')
    const errParam = c.req.query('error')
    if (errParam) {
      return c.redirect(`${process.env.NEXT_PUBLIC_APP_URL ?? 'http://localhost:3000'}/settings/integrations?stripe=denied`)
    }
    if (!code || !state) {
      throw new HTTPException(400, { message: 'Missing code or state' })
    }
    // CSRF: state must equal the current landlordId (T-2-S07)
    if (state !== landlordId) {
      throw new HTTPException(400, { message: 'Invalid state parameter' })
    }
    let token
    try {
      token = await stripeClient.oauth.token({ code, grant_type: 'authorization_code' })
    } catch (err) {
      throw new HTTPException(500, { message: `Stripe OAuth error: ${(err as Error).message}` })
    }
    const stripeUserId = token.stripe_user_id
    if (!stripeUserId) {
      throw new HTTPException(500, { message: 'Stripe did not return stripe_user_id' })
    }

    // Persist on org row (additionalField stripe_connected_account_id from Plan 02 schema)
    await withTenant(landlordId, userId, async (tx) => {
      await tx.update(organization)
        .set({ stripeConnectedAccountId: stripeUserId } as Partial<typeof organization.$inferInsert>)
        .where(eq(organization.id, landlordId))
    })

    return c.redirect(`${process.env.NEXT_PUBLIC_APP_URL ?? 'http://localhost:3000'}/settings/integrations?stripe=connected`)
  })
  // POST /disconnect — clear the connected_account_id
  .post('/disconnect', authMiddleware, requirePermission('admin'), async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    await withTenant(landlordId, userId, async (tx) => {
      await tx.update(organization)
        .set({ stripeConnectedAccountId: null } as Partial<typeof organization.$inferInsert>)
        .where(eq(organization.id, landlordId))
    })
    return c.json({ ok: true })
  })
  // GET /status — returns connection state for the integration card
  .get('/status', authMiddleware, async (c) => {
    const { landlordId } = await requireTenant(c)
    const rows = await db.select().from(organization).where(eq(organization.id, landlordId)).limit(1)
    const connected = !!(rows[0] as { stripeConnectedAccountId?: string | null })?.stripeConnectedAccountId
    return c.json({ connected })
  })
```

```tsx
"use client"

import { Elements, PaymentElement, useStripe, useElements } from "@stripe/react-stripe-js"
import type { Stripe, StripeElementsOptions } from "@stripe/stripe-js"
import { Button } from "@/modules/common/ui/button"
import { Card, CardContent } from "@/modules/common/ui/card"
import { useState } from "react"
import { toast } from "sonner"
import { useSetupIntent } from "../hooks/use-stripe-elements"

function PaymentForm({ tier, unitCount, onComplete }: { tier: 'starter' | 'growth' | 'pro'; unitCount: number; onComplete: () => void }) {
  const stripe = useStripe()
  const elements = useElements()
  const [loading, setLoading] = useState(false)

  return (
    <form
      className="flex flex-col gap-4"
      onSubmit={async (e) => {
        e.preventDefault()
        if (!stripe || !elements) return
        setLoading(true)
        try {
          const { error: confirmError, setupIntent } = await stripe.confirmSetup({
            elements,
            redirect: 'if_required',
            confirmParams: { return_url: `${window.location.origin}/settings/billing` },
          })
          if (confirmError) {
            toast.error(confirmError.message ?? "Payment method could not be saved.")
            return
          }
          const paymentMethodId = setupIntent?.payment_method as string
          if (!paymentMethodId) {
            toast.error("Could not save payment method.")
            return
          }
          const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/subscribe`, {
            method: 'POST', credentials: 'include',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({ tier, paymentMethodId, unitCount }),
          })
          if (!res.ok) {
            toast.error("Could not start subscription.")
            return
          }
          toast.success("Plan updated")
          onComplete()
        } finally { setLoading(false) }
      }}
    >
      <PaymentElement />
      <Button type="submit" size="lg" disabled={loading || !stripe || !elements}>
        Save payment method &amp; upgrade
      </Button>
    </form>
  )
}

export function PaymentMethodForm({ tier, unitCount, onComplete }: { tier: 'starter' | 'growth' | 'pro'; unitCount: number; onComplete: () => void }) {
  const { clientSecret, stripeInstance } = useSetupIntent(true)
  if (!clientSecret || !stripeInstance) {
    return <div className="text-sm text-muted-foreground">Loading payment form…</div>
  }
  const options: StripeElementsOptions = { clientSecret, appearance: { theme: 'stripe' } }
  return (
    <Card>
      <CardContent>
        <Elements stripe={stripeInstance} options={options}>
          <PaymentForm tier={tier} unitCount={unitCount} onComplete={onComplete} />
        </Elements>
      </CardContent>
    </Card>
  )
```

```tsx
"use client"

import { useState } from "react"
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter, DialogTrigger } from "@/modules/common/ui/dialog"
import { Button } from "@/modules/common/ui/button"
import { toast } from "sonner"

export type IntegrationKind = 'stripe' | 'docusign'

const COPY: Record<IntegrationKind, { title: string; body: string; cancel: string }> = {
  stripe: {
    title: "Disconnect Stripe?",
    body: "Tenant deposit payments will stop working. Disconnect only if you are switching accounts.",
    cancel: "Keep connected",
  },
  docusign: {
    title: "Disconnect DocuSign?",
    body: "You will not be able to send documents for e-signature until you reconnect.",
    cancel: "Stay connected",
  },
}

export function DisconnectIntegrationDialog({
  kind,
  trigger,
  onDisconnected,
}: {
  kind: IntegrationKind
  trigger: React.ReactNode
  onDisconnected: () => void
}) {
  const [open, setOpen] = useState(false)
  const [loading, setLoading] = useState(false)
  const copy = COPY[kind]

  return (
    <Dialog open={open} onOpenChange={setOpen}>
      <DialogTrigger asChild>{trigger}</DialogTrigger>
      <DialogContent className="sm:max-w-md">
        <DialogHeader>
          <DialogTitle>{copy.title}</DialogTitle>
          <DialogDescription>{copy.body}</DialogDescription>
        </DialogHeader>
        <DialogFooter>
          <Button variant="outline" onClick={() => setOpen(false)}>{copy.cancel}</Button>
          <Button
            variant="destructive"
            disabled={loading}
            onClick={async () => {
              setLoading(true)
              try {
                const path = kind === 'stripe' ? 'stripe-connect' : 'docusign-connect'
                const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/${path}/disconnect`, {
                  method: 'POST', credentials: 'include',
                })
                if (!res.ok) throw new Error('disconnect failed')
                toast.success("Disconnected")
                setOpen(false)
                onDisconnected()
              } catch {
                toast.error("Something went wrong. Please try again.")
              } finally { setLoading(false) }
            }}
          >
            Disconnect
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}
```

```tsx
"use client"

import { useState, useEffect } from "react"
import { IntegrationCard } from "@/modules/settings/components/integration-card"
import { Skeleton } from "@/modules/common/ui/skeleton"
import { toast } from "sonner"
import { authClient } from "@/lib/auth-client"

export default function IntegrationsPage() {
  const [stripeConnected, setStripeConnected] = useState<boolean | null>(null)
  const [docusignConnected, setDocusignConnected] = useState<boolean | null>(null)

  const refetch = async () => {
    const [s, d] = await Promise.all([
      fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/stripe-connect/status`, { credentials: 'include' }),
      fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/docusign-connect/status`, { credentials: 'include' }),
    ])
    if (s.ok) { const j = await s.json() as { connected: boolean }; setStripeConnected(j.connected) }
    if (d.ok) { const j = await d.json() as { connected: boolean }; setDocusignConnected(j.connected) }
  }
  useEffect(() => { void refetch() }, [])

  if (stripeConnected === null || docusignConnected === null) return <Skeleton className="h-32" />

  // Stripe Connect OAuth URL — landlord_id used as state for CSRF (Plan 08 Task 1 verifies state on callback)
  const handleStripeConnect = async () => {
    const { data: session } = await authClient.getSession()
    const orgId = session?.session?.activeOrganizationId
    if (!orgId) { toast.error("Could not start connection"); return }
    const callback = `${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/stripe-connect/callback`
    const url = `https://connect.stripe.com/oauth/authorize?response_type=code&client_id=${process.env.NEXT_PUBLIC_STRIPE_CLIENT_ID}&scope=read_write&redirect_uri=${encodeURIComponent(callback)}&state=${orgId}`
    window.location.href = url
  }

  const handleDocusignConnect = async () => {
    // Phase 2 stub — D-48 says UI ships now, full OAuth in Phase 4
    const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/docusign-connect/initiate`, { credentials: 'include' })
    const json = await res.json() as { url: string | null; message?: string }
    if (json.url) {
      window.location.href = json.url
    } else {
      toast.message(json.message ?? "DocuSign connection lands in Phase 4.")
    }
  }

  return (
    <div className="flex max-w-3xl flex-col gap-6">
      <h1 className="font-heading text-2xl font-medium">Integrations</h1>
      <IntegrationCard
        kind="stripe"
        name="Stripe Connect"
        description="Receive tenant deposit payments through your Stripe account."
        connected={stripeConnected}
        connectCta="Connect Stripe Account"
        onConnect={handleStripeConnect}
        onChanged={refetch}
      />
      <IntegrationCard
        kind="docusign"
        name="DocuSign"
        description="Send leases for e-signature directly from Villuh."
        connected={docusignConnected}
        connectCta="Connect DocuSign"
        onConnect={handleDocusignConnect}
        onChanged={refetch}
      />
    </div>
  )
}
```

| Browser → /api/billing/subscribe | Sends paymentMethodId (tokenized; no card data) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-2-S08 | Spoofing | Stripe Connect callback CSRF (attacker links own Stripe account) | mitigate | Callback verifies `state === landlordId` (Task 1.1 acceptance criterion + test). Without authenticated session in same browser, callback returns 401 from authMiddleware. |
| T-2-T08 | Tampering | Forged invitation acceptance | mitigate | Better Auth `acceptInvitation` validates invitation token signature + expiry; cannot accept by spoofed ID alone. |
| T-2-T09 | Tampering | Skipping payment to upgrade plan | mitigate | `/api/billing/subscribe` (Plan 05) requires real `paymentMethodId` from Stripe.confirmSetup; without a valid PM ID, Stripe rejects subscription creation. Tier change can't bypass payment. |
| T-2-I07 | Information Disclosure | Stripe Connect access_token leaks via redirect | mitigate | callback NEVER puts access_token in redirect URL or browser-visible storage. Only `stripe_user_id` is stored on org row (acceptable to expose; required for client-side Connect flows). |
| T-2-I08 | Information Disclosure | Card PAN logged | mitigate | Stripe Elements iframe — PAN never touches our servers. PCI scope reduced to SAQ A (use of fully-hosted iframe). |
| T-2-D06 | Denial of Service | OAuth callback abuse | accept | Stripe rate-limits the OAuth endpoint; Hono handles each callback in a transaction; no DOS amplification. |
| T-2-E11 | Elevation of Privilege | Manager triggers Stripe disconnect | mitigate | `/api/billing/stripe-connect/disconnect` wrapped with `requirePermission('admin')`. Only owner (or manager with admin=true override) can disconnect. |
| T-2-E12 | Elevation of Privilege | Plan limit bypass via direct API | mitigate | All write routes apply `requirePermission` AND plan-limit guards. Frontend hint via UpgradeModal (D-46) is UX layer; server enforces. |
