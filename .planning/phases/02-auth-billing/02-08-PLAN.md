**Step 1.1 — Create `packages/server/src/routes/billing/stripe-connect.ts`:**

```typescript
import { Hono } from 'hono'
import { authMiddleware, type AuthVariables } from '../../middleware/auth.js'
import { requireTenant } from '../../middleware/require-tenant.js'
import { requirePermission } from '../../middleware/require-permission.js'
import { stripeClient } from '../../auth.js'
import { db } from '@villuh/db'
import { organization } from '@villuh/db/schema/auth'
import { withTenant } from '../../lib/tenant.js'
import { eq } from 'drizzle-orm'
import { HTTPException } from 'hono/http-exception'

export const stripeConnect = new Hono<{ Variables: AuthVariables }>()
  // GET /callback — Stripe redirects here after the user authorizes the OAuth app.
  // We rely on cookie-auth (the user is signed in to Villuh in the same browser).
  .get('/callback', authMiddleware, async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    const code = c.req.query('code')
    const state = c.req.query('state')
    const errParam = c.req.query('error')
    if (errParam) {
      return c.redirect(`${process.env.NEXT_PUBLIC_APP_URL ?? 'http://localhost:3000'}/settings/integrations?stripe=denied`)
    }
    if (!code || !state) {
      throw new HTTPException(400, { message: 'Missing code or state' })
    }
    // CSRF: state must equal the current landlordId (T-2-S08)
    if (state !== landlordId) {
      throw new HTTPException(400, { message: 'Invalid state parameter' })
    }
    let token
    try {
      token = await stripeClient.oauth.token({ code, grant_type: 'authorization_code' })
    } catch (err) {
      throw new HTTPException(500, { message: `Stripe OAuth error: ${(err as Error).message}` })
    }
    const stripeUserId = token.stripe_user_id
    if (!stripeUserId) {
      throw new HTTPException(500, { message: 'Stripe did not return stripe_user_id' })
    }

    // Persist on org row (additionalField stripe_connected_account_id from Plan 02-01 schema)
    await withTenant(landlordId, userId, async (tx) => {
      await tx.update(organization)
        .set({ stripeConnectedAccountId: stripeUserId } as Partial<typeof organization.$inferInsert>)
        .where(eq(organization.id, landlordId))
    })

    return c.redirect(`${process.env.NEXT_PUBLIC_APP_URL ?? 'http://localhost:3000'}/settings/integrations?stripe=connected`)
  })
  // POST /disconnect — clear the connected_account_id (admin-gated)
  .post('/disconnect', authMiddleware, requirePermission('admin'), async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    await withTenant(landlordId, userId, async (tx) => {
      await tx.update(organization)
        .set({ stripeConnectedAccountId: null } as Partial<typeof organization.$inferInsert>)
        .where(eq(organization.id, landlordId))
    })
    return c.json({ ok: true })
  })
  // GET /status — returns connection state for the integration card
  .get('/status', authMiddleware, async (c) => {
    const { landlordId } = await requireTenant(c)
    const rows = await db.select().from(organization).where(eq(organization.id, landlordId)).limit(1)
    const connected = !!(rows[0] as { stripeConnectedAccountId?: string | null })?.stripeConnectedAccountId
    return c.json({ connected })
  })
```

| T-2-D06 | Denial of Service | OAuth callback abuse | accept | Stripe rate-limits the OAuth endpoint; Hono handles each callback in a transaction; no DOS amplification. |
