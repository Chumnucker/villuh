**Step 1.1 — Create `packages/server/src/routes/account.ts`:**

Two endpoints: PUT /api/account (update name + email — Better Auth handles email change with re-verification by default), POST /api/account/password (change password using Better Auth changePassword).

```typescript
import { Hono } from 'hono'
import { z } from 'zod'
import { zValidator } from '@hono/zod-validator'
import { authMiddleware, type AuthVariables } from '../middleware/auth.js'
import { requireTenant } from '../middleware/require-tenant.js'
import { auth } from '../auth.js'
import { HTTPException } from 'hono/http-exception'

const updateProfileSchema = z.object({
  name: z.string().min(1).max(120),
})
const changePasswordSchema = z.object({
  currentPassword: z.string().min(1),
  newPassword: z.string().min(8),
  revokeOtherSessions: z.boolean().optional(),
})

export const account = new Hono<{ Variables: AuthVariables }>()
  .put('/', authMiddleware, zValidator('json', updateProfileSchema), async (c) => {
    const { userId } = await requireTenant(c)
    const { name } = c.req.valid('json')
    // Use Better Auth's updateUser API
    try {
      await auth.api.updateUser({ body: { name }, headers: c.req.raw.headers })
    } catch (err) {
      throw new HTTPException(400, { message: (err as Error).message })
    }
    return c.json({ ok: true })
  })
  .post('/password', authMiddleware, zValidator('json', changePasswordSchema), async (c) => {
    const { currentPassword, newPassword, revokeOtherSessions } = c.req.valid('json')
    try {
      await auth.api.changePassword({
        body: { currentPassword, newPassword, revokeOtherSessions: revokeOtherSessions ?? true },
        headers: c.req.raw.headers,
      })
    } catch (err) {
      throw new HTTPException(400, { message: (err as Error).message })
    }
    return c.json({ ok: true })
  })
```

If Better Auth 1.6.9 exposes a different signature for `updateUser` or `changePassword`, adjust accordingly. The CONTRACT this plan delivers is:
- PUT /api/account with `{ name }` updates the current user's name
- POST /api/account/password with `{ currentPassword, newPassword }` rotates the password

**Step 1.2 — Create `packages/server/src/routes/audit-log.ts`:**

```typescript
import { Hono } from 'hono'
import { z } from 'zod'
import { zValidator } from '@hono/zod-validator'
import { authMiddleware, type AuthVariables } from '../middleware/auth.js'
import { requireTenant } from '../middleware/require-tenant.js'
import { withTenant } from '../lib/tenant.js'
import { sql } from 'drizzle-orm'

const querySchema = z.object({
  page: z.coerce.number().int().min(1).default(1),
  limit: z.coerce.number().int().min(1).max(100).default(25),
})

export const auditLog = new Hono<{ Variables: AuthVariables }>()
  .get('/', authMiddleware, zValidator('query', querySchema), async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    const { page, limit } = c.req.valid('query')
    const offset = (page - 1) * limit

    return withTenant(landlordId, userId, async (tx) => {
      // RLS already filters by landlord_id; we additionally filter by actor_id = userId for personal log
      const rows = await tx.execute(sql`
        SELECT id, table_name, row_id, operation, old_values, new_values, actor_id, created_at
        FROM audit_log
        WHERE actor_id = ${userId}
        ORDER BY created_at DESC
        LIMIT ${limit} OFFSET ${offset}
      `) as unknown[]
      const totalRow = await tx.execute(sql`
        SELECT COUNT(*)::int AS c FROM audit_log WHERE actor_id = ${userId}
      `) as unknown as { c: number }[]
      const total = totalRow[0]?.c ?? 0
      return c.json({ entries: rows, page, limit, total })
    })
  })
```

```tsx
"use client"

import { useState, useEffect } from "react"
import { Switch } from "@/modules/common/ui/switch"
import { Card, CardHeader, CardTitle, CardDescription, CardContent } from "@/modules/common/ui/card"
import { Skeleton } from "@/modules/common/ui/skeleton"
import { toast } from "sonner"

type Permission = { role: 'owner' | 'manager' | 'viewer'; action: 'write' | 'financial' | 'communications' | 'admin'; allowed: boolean }
const ACTIONS: Array<{ key: 'write' | 'financial' | 'communications' | 'admin'; label: string }> = [
  { key: 'write', label: 'Create / edit / delete' },
  { key: 'financial', label: 'Financial actions' },
  { key: 'communications', label: 'Communications' },
  { key: 'admin', label: 'Admin (invite, billing, settings)' },
]
const ROLES: Array<{ key: 'owner' | 'manager' | 'viewer'; label: string }> = [
  { key: 'owner', label: 'Owner' },
  { key: 'manager', label: 'Manager' },
  { key: 'viewer', label: 'Viewer' },
]

export function RolePermissionsEditor() {
  const [perms, setPerms] = useState<Permission[] | null>(null)
  const [loading, setLoading] = useState<string | null>(null)

  const refetch = async () => {
    const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/role-permissions`, { credentials: 'include' })
    if (res.ok) {
      const json = await res.json() as { permissions: Permission[] }
      setPerms(json.permissions)
    }
  }
  useEffect(() => { void refetch() }, [])

  const get = (role: Permission['role'], action: Permission['action']) => perms?.find((p) => p.role === role && p.action === action)?.allowed ?? false

  const update = async (role: Permission['role'], action: Permission['action'], allowed: boolean) => {
    const key = `${role}-${action}`
    setLoading(key)
    try {
      const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/role-permissions`, {
        method: 'PUT', credentials: 'include',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ role, action, allowed }),
      })
      if (!res.ok) throw new Error('update failed')
      await refetch()
      toast.success("Permissions updated")
    } catch {
      toast.error("Something went wrong. Please try again.")
    } finally { setLoading(null) }
  }

  if (perms === null) {
    return <Skeleton className="h-64" />
  }

  return (
    <Card>
      <CardHeader>
        <CardTitle>Role Permissions</CardTitle>
        <CardDescription>Configure what each role can do. Owner has all permissions and cannot be changed.</CardDescription>
      </CardHeader>
      <CardContent>
        <div className="grid grid-cols-[1fr_auto_auto_auto] gap-4">
          <div /> {/* corner */}
          {ROLES.map((r) => <div key={r.key} className="text-center font-medium text-sm">{r.label}</div>)}
          {ACTIONS.map((a) => (
            <>
              <div key={`${a.key}-label`} className="flex items-center text-sm">{a.label}</div>
              {ROLES.map((r) => (
                <div key={`${a.key}-${r.key}`} className="flex justify-center">
                  <Switch
                    checked={get(r.key, a.key)}
                    disabled={r.key === 'owner' || loading === `${r.key}-${a.key}`}
                    onCheckedChange={(checked) => void update(r.key, a.key, checked)}
                  />
                </div>
              ))}
            </>
          ))}
        </div>
      </CardContent>
    </Card>
  )
}
```

```tsx
    const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/team/members`, { credentials: 'include' })
    if (res.ok) {
      const json = await res.json() as { members: unknown[] }
      setMembers(json.members)
    }
  }
  useEffect(() => { void refetch() }, [])

  return (
    <div className="flex max-w-4xl flex-col gap-6">
      <h1 className="font-heading text-2xl font-medium">Team</h1>
      <InviteMemberForm onInvited={refetch} />
      {members === null ? <Skeleton className="h-32" /> : <TeamMemberList members={members as never} onChanged={refetch} />}
      <RolePermissionsEditor />
    </div>
  )
}
```
