**Step 1.1 — Create `packages/server/src/routes/account.ts`:**

Two endpoints: PUT /api/account (update name; email change handled by Better Auth with re-verification — out of scope for this plan), POST /api/account/password (change password using Better Auth `changePassword`).

```typescript
import { Hono } from 'hono'
import { z } from 'zod'
import { zValidator } from '@hono/zod-validator'
import { authMiddleware, type AuthVariables } from '../middleware/auth.js'
import { requireTenant } from '../middleware/require-tenant.js'
import { auth } from '../auth.js'
import { HTTPException } from 'hono/http-exception'

const updateProfileSchema = z.object({
  name: z.string().min(1).max(120),
})
const changePasswordSchema = z.object({
  currentPassword: z.string().min(1),
  newPassword: z.string().min(8),
  revokeOtherSessions: z.boolean().optional(),
})

export const account = new Hono<{ Variables: AuthVariables }>()
  .put('/', authMiddleware, zValidator('json', updateProfileSchema), async (c) => {
    await requireTenant(c)              // 401 if not authenticated
    const { name } = c.req.valid('json')
    try {
      await auth.api.updateUser({ body: { name }, headers: c.req.raw.headers })
    } catch (err) {
      throw new HTTPException(400, { message: (err as Error).message })
    }
    return c.json({ ok: true })
  })
  .post('/password', authMiddleware, zValidator('json', changePasswordSchema), async (c) => {
    await requireTenant(c)              // 401 if not authenticated
    const { currentPassword, newPassword, revokeOtherSessions } = c.req.valid('json')
    try {
      await auth.api.changePassword({
        body: { currentPassword, newPassword, revokeOtherSessions: revokeOtherSessions ?? true },
        headers: c.req.raw.headers,
      })
    } catch (err) {
      throw new HTTPException(400, { message: (err as Error).message })
    }
    return c.json({ ok: true })
  })
```

If Better Auth 1.6.9 exposes a different signature for `updateUser` or `changePassword`, adjust accordingly. The CONTRACT this plan delivers is:
- PUT /api/account with `{ name }` updates the current user's name
- POST /api/account/password with `{ currentPassword, newPassword }` rotates the password

**Step 2.1 — Create `packages/server/src/routes/audit-log.ts`:**

```typescript
import { Hono } from 'hono'
import { z } from 'zod'
import { zValidator } from '@hono/zod-validator'
import { authMiddleware, type AuthVariables } from '../middleware/auth.js'
import { requireTenant } from '../middleware/require-tenant.js'
import { withTenant } from '../lib/tenant.js'
import { sql } from 'drizzle-orm'

const querySchema = z.object({
  page: z.coerce.number().int().min(1).default(1),
  limit: z.coerce.number().int().min(1).max(100).default(25),
})

export const auditLog = new Hono<{ Variables: AuthVariables }>()
  .get('/', authMiddleware, zValidator('query', querySchema), async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    const { page, limit } = c.req.valid('query')
    const offset = (page - 1) * limit

    return withTenant(landlordId, userId, async (tx) => {
      // RLS already filters by landlord_id (Phase 1 tenant_isolation policy);
      // we additionally filter by actor_id = userId for personal log (SETT-02)
      const rows = await tx.execute(sql`
        SELECT id, table_name, row_id, operation, old_values, new_values, actor_id, created_at
        FROM audit_log
        WHERE actor_id = ${userId}
        ORDER BY created_at DESC
        LIMIT ${limit} OFFSET ${offset}
      `) as unknown[]
      const totalRow = await tx.execute(sql`
        SELECT COUNT(*)::int AS c FROM audit_log WHERE actor_id = ${userId}
      `) as unknown as { c: number }[]
      const total = totalRow[0]?.c ?? 0
      return c.json({ entries: rows, page, limit, total })
    })
  })
```

  it('GET /api/audit-log returns entries filtered by actor_id = current user (Phase 1 triggers)', async () => {
    await signUpTestUser('audit@test.local', 'TestPass123!', 'Audit User')
    const userId = await getCurrentUserId('audit@test.local')
    const orgId = await getOrgIdForUser(userId)
    await markUserEmailVerified(userId)
    const { cookie } = await signInTestUser('audit@test.local', 'TestPass123!')

    // Trigger an audit-loggable action: PUT /api/role-permissions (manager → admin = true)
    await fetchApi('/api/role-permissions', {
      method: 'PUT',
      headers: { 'content-type': 'application/json', cookie },
      body: JSON.stringify({ role: 'manager', action: 'admin', allowed: true }),
    })

    const res = await fetchApi('/api/audit-log', { headers: { cookie } })
    expect(res.status).toBe(200)
    const body = await res.json() as { entries: { table_name: string; actor_id: string }[]; total: number }

    expect(body.entries.every((e) => e.actor_id === userId)).toBe(true)
    // At least one should be from the role_permissions table
    expect(body.entries.some((e) => e.table_name === 'role_permissions')).toBe(true)

  it('audit log entries from another user are NOT visible (RLS + actor_id filter)', async () => {
    // User A
    await signUpTestUser('audit-a@test.local', 'TestPass123!', 'Audit A')

    const { cookie: cookieB } = await signInTestUser('audit-b@test.local', 'TestPass123!')

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → /api/account | Caller's user_id only (Better Auth identifies via session cookie) |
| Browser → /api/audit-log | actor_id filter restricts to current user; RLS restricts to org |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-2-S07 | Spoofing | Update another user's profile via /api/account | mitigate | Better Auth `updateUser` operates on the session's user only — cannot specify another user_id in the body. Acceptance criterion: route uses `headers: c.req.raw.headers`. |
| T-2-I06 | Information Disclosure | Audit log leaks across users | mitigate | Query filters by `actor_id = ${userId}` AND RLS filters by landlord_id. Acceptance test verifies cross-user isolation (Test 2). |
| T-2-R04 | Repudiation | Profile/password change unaudited | mitigate | Better Auth's `user` and `account` tables fire Phase 1 audit triggers on UPDATE; visible via /api/audit-log. |
| T-2-D05 | Denial of Service | Audit log query unbounded | mitigate | Query enforces `LIMIT ${limit}` clamped to 100; default 25. Pagination prevents full-table reads. Acceptance test verifies pagination contract. |
| T-2-T07 | Tampering | Caller bypasses currentPassword check | mitigate | Better Auth `changePassword` validates `currentPassword` against stored hash; we never see the password. |
</threat_model>
