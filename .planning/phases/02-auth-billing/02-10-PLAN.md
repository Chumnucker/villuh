**Step 1.1 — Create `packages/web/modules/settings/components/invite-member-form.tsx`:**

Per UI-SPEC §Settings/Team Invite member: Card with email Input + role Select + "Send invite" button. On 402 PLAN_LIMIT (D-46), show explicit toast — Plan 02-12's UpgradeModalProvider can also catch this when wired.

```tsx
"use client"

import { useState } from "react"
import { Button } from "@/modules/common/ui/button"
import { Input } from "@/modules/common/ui/input"
import { Field, FieldLabel } from "@/modules/common/ui/field"
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/modules/common/ui/select"
import { Card, CardHeader, CardTitle, CardContent, CardFooter } from "@/modules/common/ui/card"
import { toast } from "sonner"

export function InviteMemberForm({ onInvited }: { onInvited: () => void }) {
  const [email, setEmail] = useState("")
  const [role, setRole] = useState<'manager' | 'viewer' | 'owner'>('viewer')
  const [loading, setLoading] = useState(false)

  return (
    <Card>
      <CardHeader>
        <CardTitle>Invite a team member</CardTitle>
      </CardHeader>
      <form onSubmit={async (e) => {
        e.preventDefault()
        setLoading(true)
        try {
          const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/team/invitations`, {
            method: 'POST', credentials: 'include',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({ email, role }),
          })
          if (res.status === 402) {
            const body = await res.json() as { code?: string; message?: string }
            const text = (body.message ?? JSON.stringify(body))
            if (text.includes('PLAN_LIMIT')) {
              toast.error("You've reached your team member limit. Upgrade your plan.")
              return
            }
          }
          if (!res.ok) throw new Error('invite failed')
          toast.success("Invitation sent")
          setEmail("")
          onInvited()
        } catch {
          toast.error("Something went wrong. Please try again.")
        } finally { setLoading(false) }
      }}>
        <CardContent className="flex flex-col gap-4 sm:flex-row sm:items-end">
          <Field className="flex-1"><FieldLabel>Email</FieldLabel><Input type="email" value={email} onChange={(e) => setEmail(e.target.value)} required /></Field>
          <Field>
            <FieldLabel>Role</FieldLabel>
            <Select value={role} onValueChange={(v) => setRole(v as 'manager' | 'viewer' | 'owner')}>
              <SelectTrigger className="w-32"><SelectValue /></SelectTrigger>
              <SelectContent>
                <SelectItem value="owner">Owner</SelectItem>
                <SelectItem value="manager">Manager</SelectItem>
                <SelectItem value="viewer">Viewer</SelectItem>
              </SelectContent>
            </Select>
          </Field>
        </CardContent>
        <CardFooter className="justify-end">
          <Button type="submit" disabled={loading}>Send invite</Button>
        </CardFooter>
      </form>
    </Card>
  )
}
```

```tsx
"use client"

import { Table, TableHeader, TableBody, TableHead, TableRow, TableCell } from "@/modules/common/ui/table"
import { Avatar, AvatarFallback, AvatarImage } from "@/modules/common/ui/avatar"
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/modules/common/ui/select"

type Member = { member_id: string; user_id: string; name: string; email: string; image: string | null; role: 'owner' | 'manager' | 'viewer'; joined_at: string }

export function TeamMemberList({ members, onChanged }: { members: Member[]; onChanged: () => void }) {
  if (members.length === 0) {
    return (
      <Card>
        <CardContent className="flex flex-col items-center gap-2 py-12 text-center">

      <TableHeader>
        <TableRow>

          <TableHead>Email</TableHead>

          <TableHead>Joined</TableHead>
          <TableHead></TableHead>
        </TableRow>
      </TableHeader>
      <TableBody>
        {members.map((m) => (
          <TableRow key={m.member_id}>
            <TableCell>
              <div className="flex items-center gap-2">

              </div>
            </TableCell>
            <TableCell className="text-muted-foreground">{m.email}</TableCell>
```

**Step 2.2 — Create `packages/web/modules/settings/components/role-permissions-editor.tsx`:**

Per UI-SPEC: Card with "Role Permissions" + Switch grid per role × action (3 roles × 4 actions). Owner column read-only / disabled (since defaults are immutable per D-09 and Plan 02-04 PUT rejects owner edits).

```tsx
"use client"

import { useState, useEffect } from "react"
import { Switch } from "@/modules/common/ui/switch"
import { Card, CardHeader, CardTitle, CardDescription, CardContent } from "@/modules/common/ui/card"
import { Skeleton } from "@/modules/common/ui/skeleton"
import { toast } from "sonner"

type Permission = { role: 'owner' | 'manager' | 'viewer'; action: 'write' | 'financial' | 'communications' | 'admin'; allowed: boolean }
const ACTIONS: Array<{ key: 'write' | 'financial' | 'communications' | 'admin'; label: string }> = [
  { key: 'write', label: 'Create / edit / delete' },
  { key: 'financial', label: 'Financial actions' },
  { key: 'communications', label: 'Communications' },
  { key: 'admin', label: 'Admin (invite, billing, settings)' },
]
const ROLES: Array<{ key: 'owner' | 'manager' | 'viewer'; label: string }> = [
  { key: 'owner', label: 'Owner' },
  { key: 'manager', label: 'Manager' },
  { key: 'viewer', label: 'Viewer' },
]

export function RolePermissionsEditor() {
  const [perms, setPerms] = useState<Permission[] | null>(null)
  const [loading, setLoading] = useState<string | null>(null)

  const refetch = async () => {
    const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/role-permissions`, { credentials: 'include' })
    if (res.ok) {
      const json = await res.json() as { permissions: Permission[] }
      setPerms(json.permissions)
    }
  }
  useEffect(() => { void refetch() }, [])

  const get = (role: Permission['role'], action: Permission['action']) => perms?.find((p) => p.role === role && p.action === action)?.allowed ?? false

  const update = async (role: Permission['role'], action: Permission['action'], allowed: boolean) => {
    const key = `${role}-${action}`
    setLoading(key)
    try {
      const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/role-permissions`, {
        method: 'PUT', credentials: 'include',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ role, action, allowed }),
      })
      if (!res.ok) throw new Error('update failed')
      await refetch()
      toast.success("Permissions updated")
    } catch {
      toast.error("Something went wrong. Please try again.")
    } finally { setLoading(null) }
  }

  if (perms === null) {
    return <Skeleton className="h-64" />
  }

  return (
    <Card>
      <CardHeader>
        <CardTitle>Role Permissions</CardTitle>
        <CardDescription>Configure what each role can do. Owner has all permissions and cannot be changed.</CardDescription>
      </CardHeader>
      <CardContent>
        <div className="grid grid-cols-[1fr_auto_auto_auto] gap-4">
          <div /> {/* corner */}
          {ROLES.map((r) => <div key={r.key} className="text-center font-medium text-sm">{r.label}</div>)}
          {ACTIONS.map((a) => (
            <>
              <div key={`${a.key}-label`} className="flex items-center text-sm">{a.label}</div>
              {ROLES.map((r) => (
                <div key={`${a.key}-${r.key}`} className="flex justify-center">
                  <Switch
                    checked={get(r.key, a.key)}
                    disabled={r.key === 'owner' || loading === `${r.key}-${a.key}`}
                    onCheckedChange={(checked) => void update(r.key, a.key, checked)}
                  />
                </div>
              ))}
            </>
          ))}
        </div>
      </CardContent>
    </Card>
  )
}
```
