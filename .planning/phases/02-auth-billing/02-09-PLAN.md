**Step 2.1 — Create `packages/web/modules/settings/components/account-form.tsx`:**

Per UI-SPEC §Account: Profile section in a Card with Name + Email fields + "Save changes" button. Email is read-only because Better Auth handles email change with re-verification (deferred).

```tsx
"use client"

import { useState, useEffect } from "react"
import { Button } from "@/modules/common/ui/button"
import { Input } from "@/modules/common/ui/input"
import { Field, FieldLabel } from "@/modules/common/ui/field"
import { Card, CardHeader, CardTitle, CardDescription, CardContent, CardFooter } from "@/modules/common/ui/card"
import { authClient } from "@/lib/auth-client"
import { toast } from "sonner"

export function AccountForm() {
  const { data: session } = authClient.useSession()
  const [name, setName] = useState("")
  const [loading, setLoading] = useState(false)

  useEffect(() => { if (session?.user?.name) setName(session.user.name) }, [session?.user?.name])

  return (
    <Card>
      <CardHeader>
        <CardTitle>Profile</CardTitle>
        <CardDescription>Your name and email displayed to your team.</CardDescription>
      </CardHeader>
      <form onSubmit={async (e) => {
        e.preventDefault()
        setLoading(true)
        try {
          const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/account`, {
            method: 'PUT', credentials: 'include',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({ name }),
          })
          if (!res.ok) throw new Error('save failed')
          toast.success("Profile updated")
        } catch {
          toast.error("Something went wrong. Please try again.")
        } finally { setLoading(false) }
      }}>
        <CardContent className="flex flex-col gap-4">
          <Field><FieldLabel>Name</FieldLabel><Input value={name} onChange={(e) => setName(e.target.value)} required /></Field>
          <Field><FieldLabel>Email</FieldLabel><Input value={session?.user?.email ?? ""} readOnly disabled /></Field>
        </CardContent>
        <CardFooter className="justify-end">
          <Button type="submit" disabled={loading}>Save changes</Button>
        </CardFooter>
      </form>
    </Card>
  )
}
```

```tsx
"use client"

import { useState } from "react"
import { Button } from "@/modules/common/ui/button"
import { PasswordInput } from "@/modules/common/ui/input"
import { Field, FieldLabel, FieldError } from "@/modules/common/ui/field"
import { Card, CardHeader, CardTitle, CardDescription, CardContent, CardFooter } from "@/modules/common/ui/card"
import { toast } from "sonner"

export function PasswordChangeForm() {
  const [current, setCurrent] = useState("")
  const [next, setNext] = useState("")
  const [confirm, setConfirm] = useState("")
  const [loading, setLoading] = useState(false)
  const [mismatch, setMismatch] = useState(false)

  return (
    <Card>
      <CardHeader>
        <CardTitle>Password</CardTitle>
        <CardDescription>Change your password. You'll be signed out of other sessions.</CardDescription>
      </CardHeader>
      <form onSubmit={async (e) => {
        e.preventDefault()
        if (next !== confirm) { setMismatch(true); return }
        setMismatch(false)
        setLoading(true)
        try {
          const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/account/password`, {
            method: 'POST', credentials: 'include',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({ currentPassword: current, newPassword: next, revokeOtherSessions: true }),
          })
          if (!res.ok) {
            toast.error("Incorrect current password.")
            return
          }
          toast.success("Password updated")
          setCurrent(""); setNext(""); setConfirm("")
        } finally { setLoading(false) }
      }}>
        <CardContent className="flex flex-col gap-4">
          <Field><FieldLabel>Current password</FieldLabel><PasswordInput value={current} onChange={(e) => setCurrent(e.target.value)} required autoComplete="current-password" /></Field>
          <Field><FieldLabel>New password</FieldLabel><PasswordInput value={next} onChange={(e) => setNext(e.target.value)} required minLength={8} autoComplete="new-password" /></Field>
          <Field>
            <FieldLabel>Confirm new password</FieldLabel>
            <PasswordInput value={confirm} onChange={(e) => setConfirm(e.target.value)} required minLength={8} autoComplete="new-password" aria-invalid={mismatch || undefined} />
            {mismatch && <FieldError>Passwords do not match</FieldError>}
          </Field>
        </CardContent>
        <CardFooter className="justify-end">
          <Button type="submit" disabled={loading}>Save changes</Button>
        </CardFooter>
      </form>
    </Card>
  )
}
```

**Step 3.1 — Create `packages/web/modules/settings/components/audit-log-table.tsx`:**

Per UI-SPEC §Audit Log: Table with Timestamp, Action, Resource, Details. Pagination. Empty state.

```tsx
"use client"

import { useState, useEffect } from "react"
import { Table, TableHeader, TableBody, TableHead, TableRow, TableCell } from "@/modules/common/ui/table"
import { Pagination, PaginationContent, PaginationItem, PaginationLink, PaginationNext, PaginationPrevious } from "@/modules/common/ui/pagination"
import { Skeleton } from "@/modules/common/ui/skeleton"
import { Card, CardContent } from "@/modules/common/ui/card"

type Entry = {
  id: string
  table_name: string
  row_id: string
  operation: 'INSERT' | 'UPDATE' | 'DELETE'
  old_values: unknown
  new_values: unknown
  actor_id: string
  created_at: string
}

export function AuditLogTable() {
  const [entries, setEntries] = useState<Entry[] | null>(null)
  const [page, setPage] = useState(1)
  const [total, setTotal] = useState(0)
  const limit = 25

  useEffect(() => {
    let active = true
    void (async () => {
      const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/audit-log?page=${page}&limit=${limit}`, { credentials: 'include' })
      if (!res.ok) return
      const json = await res.json() as { entries: Entry[]; total: number }
      if (active) { setEntries(json.entries); setTotal(json.total) }
    })()
    return () => { active = false }
  }, [page])

  if (entries === null) {
    return <div className="flex flex-col gap-2"><Skeleton className="h-10" /><Skeleton className="h-10" /><Skeleton className="h-10" /></div>
  }

  if (entries.length === 0) {
    return (
      <Card>
        <CardContent className="flex flex-col items-center gap-2 py-12 text-center">
          <h3 className="font-heading text-base font-medium">No activity yet</h3>
          <p className="text-sm text-muted-foreground">Account activity will appear here as you use Villuh.</p>
        </CardContent>
      </Card>
    )
  }

  const totalPages = Math.max(1, Math.ceil(total / limit))

  return (
    <div className="flex flex-col gap-4">
      <Table>
        <TableHeader>
          <TableRow>
            <TableHead>Timestamp</TableHead>
            <TableHead>Action</TableHead>
            <TableHead>Resource</TableHead>
            <TableHead>Details</TableHead>
          </TableRow>
        </TableHeader>
        <TableBody>
          {entries.map((e) => (
            <TableRow key={e.id}>
              <TableCell>{new Date(e.created_at).toLocaleString()}</TableCell>
              <TableCell>{e.operation}</TableCell>
              <TableCell>{e.table_name}</TableCell>
              <TableCell className="max-w-md truncate text-muted-foreground">{e.row_id}</TableCell>
            </TableRow>
          ))}
        </TableBody>
      </Table>
      {totalPages > 1 && (
        <Pagination>
          <PaginationContent>
            <PaginationItem>
              <PaginationPrevious onClick={() => setPage((p) => Math.max(1, p - 1))} />
            </PaginationItem>
            <PaginationItem>

      )}
    </div>
  )
}
```
