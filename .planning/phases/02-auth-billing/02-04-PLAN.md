**Step 1.1 — Create `packages/server/src/lib/permissions.ts`:**

```typescript
import { db } from '@villuh/db'
import { rolePermissions } from '@villuh/db/schema/role-permissions'
import { and, eq } from 'drizzle-orm'
import { withTenant } from './tenant.js'

export const ACTION_CATEGORIES = ['write', 'financial', 'communications', 'admin'] as const
export type ActionCategory = (typeof ACTION_CATEGORIES)[number]

export const ROLES = ['owner', 'manager', 'viewer'] as const
export type Role = (typeof ROLES)[number]

/**
 * Default permission sets per D-09.
 * Owner: all true.
 * Manager: writes + financial + comms (no admin).
 * Viewer: read-only (all false).
 */
export const DEFAULT_PERMISSIONS: Record<Role, Record<ActionCategory, boolean>> = {
  owner:   { write: true,  financial: true,  communications: true,  admin: true  },
  manager: { write: true,  financial: true,  communications: true,  admin: false },
  viewer:  { write: false, financial: false, communications: false, admin: false },
}

/**
 * Seed the 12 default permission rows for a new org (3 roles × 4 actions).
 * Uses owner-context withTenant — system actor — and ON CONFLICT DO NOTHING for idempotency.
 * Called from Better Auth org plugin afterCreate hook.
 */
export async function seedDefaultPermissions(orgId: string): Promise<void> {
  const rows: Array<{ landlordId: string; role: Role; action: ActionCategory; allowed: boolean }> = []
  for (const role of ROLES) {
    for (const action of ACTION_CATEGORIES) {
      rows.push({
        landlordId: orgId,
        role,
        action,
        allowed: DEFAULT_PERMISSIONS[role][action],
      })
    }
  }
  await withTenant(orgId, 'system', async (tx) => {
    for (const row of rows) {
      await tx.insert(rolePermissions).values(row).onConflictDoNothing({ target: [rolePermissions.landlordId, rolePermissions.role, rolePermissions.action] })
    }
  })
}

/**
 * Load the permission map for a single role within an org.
 * Returns { write, financial, communications, admin }.
 * If a row is missing for any category, fall back to DEFAULT_PERMISSIONS for that role.
 */
export async function loadRolePermissions(orgId: string, role: Role): Promise<Record<ActionCategory, boolean>> {
  return withTenant(orgId, 'system', async (tx) => {
    const rows = await tx.select().from(rolePermissions).where(and(
      eq(rolePermissions.landlordId, orgId),
      eq(rolePermissions.role, role),
    ))
    const result: Record<ActionCategory, boolean> = { ...DEFAULT_PERMISSIONS[role] }
    for (const row of rows) {
      if (ACTION_CATEGORIES.includes(row.action as ActionCategory)) {
        result[row.action as ActionCategory] = row.allowed
      }
    }
    return result
  })
}

/**
 * Check whether a given role is allowed a given action.
 * Owner short-circuits to true (D-09).
 */
export async function isAllowed(orgId: string, role: Role, action: ActionCategory): Promise<boolean> {
  if (role === 'owner') return true
  const perms = await loadRolePermissions(orgId, role)
  return perms[action] ?? false
}
```

**Step 1.2 — Create `packages/server/src/middleware/require-permission.ts`:**

```typescript
import { createMiddleware } from 'hono/factory'
import { HTTPException } from 'hono/http-exception'
import type { AuthVariables } from './auth.js'
import { auth } from '../auth.js'
import { isAllowed, type ActionCategory, type Role } from '../lib/permissions.js'

/**
 * Factory: returns Hono middleware that 403s if the active member's role lacks the given action permission.
 * Owner role always passes (D-09).
 *
 * Usage:
 *   app.post('/api/properties', authMiddleware, requirePermission('write'), handler)
 */
export function requirePermission(action: ActionCategory) {
  return createMiddleware<{ Variables: AuthVariables }>(async (c, next) => {
    const user = c.get('user')
    const session = c.get('session')
    if (!user || !session) throw new HTTPException(401, { message: 'Unauthorized' })

    const orgId = (session as { activeOrganizationId?: string }).activeOrganizationId
    if (!orgId) throw new HTTPException(401, { message: 'No active organization' })

    // Look up the active member's role via Better Auth org API
    const member = await auth.api.getActiveMember({ headers: c.req.raw.headers }).catch(() => null)
    const role = (member as { role?: Role } | null)?.role
    if (!role) throw new HTTPException(403, { message: 'Not a member of this organization' })

    const allowed = await isAllowed(orgId, role, action)
    if (!allowed) throw new HTTPException(403, { message: `Forbidden: ${action} permission required` })

    await next()
  })
}

import { Hono } from 'hono'
import { authMiddleware, type AuthVariables } from '../middleware/auth.js'
import { requireTenant } from '../middleware/require-tenant.js'
import { db } from '@villuh/db'
import { organization, member } from '@villuh/db/schema/auth'
import { planLimits } from '@villuh/db/schema/plan-limits'
import { loadRolePermissions, type Role } from '../lib/permissions.js'
import { auth } from '../auth.js'
import { eq, and, sql } from 'drizzle-orm'
import { withTenant } from '../lib/tenant.js'

export const me = new Hono<{ Variables: AuthVariables }>()
  .get('/', authMiddleware, async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    const user = c.get('user')!

    return withTenant(landlordId, userId, async (tx) => {
      // Load org row (denormalized subscription_tier, subscription_status, setup_completed per D-23)
      const orgRow = await tx.select().from(organization).where(eq(organization.id, landlordId)).limit(1)
      if (orgRow.length === 0) {
        return c.json({ error: 'Organization not found' }, 404)
      }
      const org = orgRow[0]
      const tier = (org.subscriptionTier as string) ?? 'trial'
      const status = (org.subscriptionStatus as string) ?? 'trialing'

      // Member count
      const memberCountRow = await tx.execute(sql`SELECT COUNT(*)::int AS c FROM "member" WHERE organization_id = ${landlordId}`)
      const member_count = ((memberCountRow as unknown as { c: number }[])[0] ?? { c: 0 }).c

      // Lookup limits from plan_limits — tier 'trial' uses 'pro' limits per D-19 (full Pro access)
      const lookupTier = tier === 'trial' ? 'pro' : tier
      const limitRows = await tx.select().from(planLimits).where(eq(planLimits.tier, lookupTier))
      const limitsMap: Record<string, number> = {}
      for (const r of limitRows) {
        limitsMap[r.resource] = r.limit
      }

      // Resolve user's role (member.role)
      const memberRows = await tx.execute(sql`SELECT role FROM "member" WHERE organization_id = ${landlordId} AND user_id = ${userId} LIMIT 1`) as unknown as { role: Role }[]
      const role: Role = memberRows[0]?.role ?? 'viewer'
      const permissions = role === 'owner'
        ? { write: true, financial: true, communications: true, admin: true }
        : await loadRolePermissions(landlordId, role)

      return c.json({
        user: { id: user.id, name: user.name, email: user.email, image: (user as { image?: string }).image ?? null },
        org: {
          id: org.id,
          name: org.name,
          slug: org.slug,
          subscription_tier: tier,
          subscription_status: status,
          setup_completed: (org as { setupCompleted?: boolean; setup_completed?: boolean }).setupCompleted ?? (org as { setup_completed?: boolean }).setup_completed ?? false,
        },
        tier,
        // Phase 3 will populate unit_count and property_count from real tables;
        // for Phase 2, these are 0 until Phase 3 ships properties/units schema.
        unit_count: 0,
        unit_limit: limitsMap.units ?? 5,
        property_count: 0,
        property_limit: limitsMap.properties ?? 1,
        member_count,
        member_limit: limitsMap.members ?? 2,
        subscription_status: status,
        permissions,
      })
    })
  })

import { Hono } from 'hono'
import { z } from 'zod'
import { zValidator } from '@hono/zod-validator'
import { authMiddleware, type AuthVariables } from '../middleware/auth.js'
import { requireTenant } from '../middleware/require-tenant.js'
import { requirePermission } from '../middleware/require-permission.js'
import { auth } from '../auth.js'
import { db } from '@villuh/db'
import { planLimits } from '@villuh/db/schema/plan-limits'
import { organization } from '@villuh/db/schema/auth'
import { withTenant } from '../lib/tenant.js'
import { eq, sql } from 'drizzle-orm'
import { HTTPException } from 'hono/http-exception'

const inviteSchema = z.object({
  email: z.string().email(),
  role: z.enum(['owner', 'manager', 'viewer']),
})
const updateRoleSchema = z.object({
  role: z.enum(['owner', 'manager', 'viewer']),
})

export const team = new Hono<{ Variables: AuthVariables }>()
  .get('/members', authMiddleware, async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    return withTenant(landlordId, userId, async (tx) => {
      const rows = await tx.execute(sql`
        SELECT m.id AS member_id, m.role, u.id AS user_id, u.name, u.email, u.image, m."createdAt" AS joined_at
        FROM "member" m
        JOIN "user" u ON u.id = m.user_id
        WHERE m.organization_id = ${landlordId}
        ORDER BY m."createdAt" ASC
      `)
      return c.json({ members: rows as unknown[] })
    })
  })
  .get('/invitations', authMiddleware, async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    return withTenant(landlordId, userId, async (tx) => {
      const rows = await tx.execute(sql`
        SELECT id, email, role, status, "expiresAt" AS expires_at, "inviterId" AS inviter_id
        FROM "invitation"
        WHERE organization_id = ${landlordId} AND status = 'pending'
        ORDER BY "expiresAt" DESC
      `)
      return c.json({ invitations: rows as unknown[] })
    })
  })
  .post(
    '/invitations',
    authMiddleware,
    requirePermission('admin'),
    zValidator('json', inviteSchema),
    async (c) => {
      const { landlordId, userId } = await requireTenant(c)
      const { email, role } = c.req.valid('json')

      // Plan limit check on members (member_limit)
      await withTenant(landlordId, userId, async (tx) => {
        const memberCountRow = await tx.execute(sql`
          SELECT (
            (SELECT COUNT(*) FROM "member" WHERE organization_id = ${landlordId})
            + (SELECT COUNT(*) FROM "invitation" WHERE organization_id = ${landlordId} AND status = 'pending')
          )::int AS c
        `) as unknown as { c: number }[]
        const totalMembers = memberCountRow[0]?.c ?? 0

        const orgRow = await tx.select().from(organization).where(eq(organization.id, landlordId)).limit(1)
        const tier = (orgRow[0]?.subscriptionTier as string) ?? 'trial'
        const lookupTier = tier === 'trial' ? 'pro' : tier
        const limitRow = await tx.select().from(planLimits).where(sql`tier = ${lookupTier} AND resource = 'members'`).limit(1)
        const limit = limitRow[0]?.limit ?? 2

        if (limit !== -1 && totalMembers >= limit) {
          throw new HTTPException(402, {
            message: JSON.stringify({
              code: 'PLAN_LIMIT',
              resource: 'members',
              limit,
              current: totalMembers,
              upgrade_to: tier === 'starter' ? 'growth' : 'pro',
            }),
          })
        }
      })

      // Better Auth invitation creation — fires sendInvitationEmail callback (Plan 02 wired Resend)
      const invitation = await auth.api.createInvitation({
        body: { email, role, organizationId: landlordId },
        headers: c.req.raw.headers,
      })
      return c.json({ invitation })
    }
  )
  .delete('/members/:userId', authMiddleware, requirePermission('admin'), async (c) => {
    const { landlordId } = await requireTenant(c)
    const targetUserId = c.req.param('userId')
    await auth.api.removeMember({
      body: { memberIdOrEmail: targetUserId, organizationId: landlordId },
      headers: c.req.raw.headers,
    })
    return c.json({ ok: true })
  })
  .put(
    '/members/:userId/role',
    authMiddleware,
    requirePermission('admin'),
    zValidator('json', updateRoleSchema),
    async (c) => {
      const { landlordId } = await requireTenant(c)
      const targetUserId = c.req.param('userId')
      const { role } = c.req.valid('json')
      // Better Auth API — verify exact signature in 1.6.9
      await auth.api.updateMemberRole({
        body: { memberId: targetUserId, role, organizationId: landlordId },
        headers: c.req.raw.headers,
      })
      return c.json({ ok: true })
    }
  )
```

import { Hono } from 'hono'
import { z } from 'zod'
import { zValidator } from '@hono/zod-validator'
import { authMiddleware, type AuthVariables } from '../middleware/auth.js'
import { requireTenant } from '../middleware/require-tenant.js'
import { requirePermission } from '../middleware/require-permission.js'
import { rolePermissions } from '@villuh/db/schema/role-permissions'
import { withTenant } from '../lib/tenant.js'
import { ACTION_CATEGORIES, ROLES, DEFAULT_PERMISSIONS } from '../lib/permissions.js'
import { and, eq } from 'drizzle-orm'

const updateSchema = z.object({
  role: z.enum(['owner', 'manager', 'viewer']),
  action: z.enum(['write', 'financial', 'communications', 'admin']),
  allowed: z.boolean(),
})

export const rolePerms = new Hono<{ Variables: AuthVariables }>()
  .get('/', authMiddleware, async (c) => {
    const { landlordId, userId } = await requireTenant(c)
    return withTenant(landlordId, userId, async (tx) => {
      const rows = await tx.select().from(rolePermissions).where(eq(rolePermissions.landlordId, landlordId))
      return c.json({ permissions: rows })
    })
  })
  .put(
    '/',
    authMiddleware,
    requirePermission('admin'),
    zValidator('json', updateSchema),
    async (c) => {
      const { landlordId, userId } = await requireTenant(c)
      const { role, action, allowed } = c.req.valid('json')

      // D-09: owner permissions are not configurable (always all true). Reject changes targeting owner.
      if (role === 'owner') {
        return c.json({ error: 'Owner permissions are not configurable' }, 400)
      }

      await withTenant(landlordId, userId, async (tx) => {
        await tx
          .insert(rolePermissions)
          .values({ landlordId, role, action, allowed })
          .onConflictDoUpdate({
            target: [rolePermissions.landlordId, rolePermissions.role, rolePermissions.action],
            set: { allowed, updatedAt: new Date() },
          })
      })
      return c.json({ ok: true })
    }
  )
