**Step 1.1 — Create `packages/server/src/lib/email.ts`:**

Singleton Resend client + three send helpers. All three call Resend with React-email JSX templates rendered to HTML. They are async functions but called as `void send...()` from Better Auth callbacks (fire-and-forget) so failures don't block the request.

```typescript
import { Resend } from 'resend'
import { render } from '@react-email/components'
import { VerifyEmail } from './email-templates/verify-email.js'
import { ResetPassword } from './email-templates/reset-password.js'
import { Invitation } from './email-templates/invitation.js'

if (!process.env.RESEND_API_KEY) {
  throw new Error('RESEND_API_KEY is required')
}
if (!process.env.RESEND_FROM_EMAIL) {
  throw new Error('RESEND_FROM_EMAIL is required (must be a verified domain or onboarding@resend.dev for dev)')
}

export const resend = new Resend(process.env.RESEND_API_KEY)
const FROM = process.env.RESEND_FROM_EMAIL

export async function sendVerificationEmail(to: string, verifyUrl: string): Promise<void> {
  const html = await render(VerifyEmail({ verifyUrl }))
  await resend.emails.send({
    from: FROM,
    to,
    subject: 'Verify your Villuh email',
    html,
  })
}

export async function sendResetPasswordEmail(to: string, resetUrl: string): Promise<void> {
  const html = await render(ResetPassword({ resetUrl }))
  await resend.emails.send({
    from: FROM,
    to,
    subject: 'Reset your Villuh password',
    html,
  })
}

export async function sendInvitationEmail(
  to: string,
  inviteUrl: string,
  inviterName: string,
  orgName: string,
  role: string
): Promise<void> {
  const html = await render(Invitation({ inviteUrl, inviterName, orgName, role }))
  await resend.emails.send({
    from: FROM,
    to,
    subject: `${inviterName} invited you to join ${orgName} on Villuh`,
    html,
  })
}
```

**Step 2.1 — Create `packages/server/src/lib/billing.ts`:**

Helpers used by Better Auth Stripe plugin callbacks (`onSubscriptionCreated`, `onSubscriptionUpdate`, `onSubscriptionDeleted`, `onEvent`). They UPDATE the `organization` row's denormalized fields (per D-23). All DB writes go through `withTenant` so audit_log triggers fire under a real actor; the actor for system events is `'system'` (the Phase 1 sentinel).

```typescript
import { db } from '@villuh/db'
import { organization } from '@villuh/db/schema/auth'
import { eq, sql } from 'drizzle-orm'
import { withTenant } from './tenant.js'

/**
 * Sync subscription_tier + subscription_status on the org row.
 * Called from Better Auth Stripe plugin onSubscriptionCreated and onSubscriptionUpdate.
 * referenceId = organization.id (Better Auth Stripe plugin convention when organization billing is enabled).
 */
export async function syncOrgSubscriptionTier(
  orgId: string,
  tier: string,                      // 'starter' | 'growth' | 'pro' | 'trial'
  status: string                     // 'trialing' | 'active' | 'past_due' | 'canceled' | 'expired' | 'payment_failed'
): Promise<void> {
  await withTenant(orgId, 'system', async (tx) => {
    await tx
      .update(organization)
      .set({ subscriptionTier: tier, subscriptionStatus: status })
      .where(eq(organization.id, orgId))
  })
}

/**
 * Set only the subscription_status column.
 * Used by onSubscriptionDeleted (status = 'expired') and onEvent invoice.payment_failed.
 */
export async function setOrgSubscriptionStatus(orgId: string, status: string): Promise<void> {
  await withTenant(orgId, 'system', async (tx) => {
    await tx
      .update(organization)
      .set({ subscriptionStatus: status })
      .where(eq(organization.id, orgId))
  })
}

/**
 * Mark setup_completed = true on the org row. Called by /api/billing/trial-start once trial is created (Plan 06 wires this from wizard step 4).
 */
export async function markSetupCompleted(orgId: string): Promise<void> {
  await withTenant(orgId, 'system', async (tx) => {
    await tx
      .update(organization)
      .set({ setupCompleted: true })
      .where(eq(organization.id, orgId))
  })
}

/**
 * Stripe customer creation helper used by Plan 05 trial-start route.
 * Returns existing customer ID if Better Auth Stripe plugin already created one (it stores it on the customer table).
 * Otherwise creates a new customer and returns the ID.
 *
 * Note: Better Auth Stripe plugin with createCustomerOnSignUp=false leaves customer creation to us;
 * Plan 05 will create the customer immediately before calling stripe.subscriptions.create().
 */
export async function getOrCreateStripeCustomer(
  stripeClient: import('stripe').default,
  orgId: string,
  ownerEmail: string
): Promise<string> {
  // Stub: Plan 05 will implement full lookup against Better Auth's stripe_customer table
  // For now, search by metadata.landlord_id then create if missing.
  const existing = await stripeClient.customers.search({
    query: `metadata['landlord_id']:'${orgId}'`,
    limit: 1,
  })
  if (existing.data.length > 0) return existing.data[0].id

  const created = await stripeClient.customers.create({
    email: ownerEmail,
    metadata: { landlord_id: orgId },
  })
  return created.id
}

/**
 * Handle invoice.payment_failed Stripe event.
 * Per D-22: notify landlord in-app + start grace period. Phase 2 hard-blocks on expired (Deferred Ideas: grace period deferred);
 * for now, set subscription_status = 'payment_failed' which the gate middleware does NOT block on (only 'expired' blocks).
 * The gate behavior is finalized in Plan 06.
 */
```

```typescript
import type { Context } from 'hono'
import { HTTPException } from 'hono/http-exception'
import type { AuthVariables } from './auth.js'

/**
 * Resolve landlordId from Better Auth session for use with withTenant.
 * landlordId = session.activeOrganizationId per CLAUDE.md (NOT user.id — RESEARCH.md Pitfall 7).

export async function requireTenant(c: Context<{ Variables: AuthVariables }>): Promise<{ landlordId: string; userId: string }> {
  const user = c.get('user')

}
```
