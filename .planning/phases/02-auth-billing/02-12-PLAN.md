```tsx
"use client"

import { useState } from "react"
import { LinkButton, Button } from "@/modules/common/ui/button"
import { PlatformIcon } from "@/modules/common/ui/icon"
import type { BillingStatus } from "../hooks/use-billing-status"

function pctOf(current: number, limit: number) {
  if (limit === -1) return 0
  return current / Math.max(1, limit)
}

export function UsageWarningBanner({ status }: { status: BillingStatus | null }) {
  const [dismissed, setDismissed] = useState(false)
  if (!status || dismissed) return null

  const items = [
    { name: 'units' as const, label: 'units', current: status.unit_count, limit: status.unit_limit },
    { name: 'properties' as const, label: 'properties', current: status.property_count, limit: status.property_limit },
    { name: 'members' as const, label: 'team members', current: status.member_count, limit: status.member_limit },
  ]
  const triggered = items.find((it) => pctOf(it.current, it.limit) >= 0.8)
  if (!triggered) return null

  return (
    <div className="flex h-10 w-full items-center justify-between border-b border-amber-200 bg-amber-50 px-4 text-sm text-amber-900 dark:border-amber-800 dark:bg-amber-950/30 dark:text-amber-200">
      <div>
        You're using {triggered.current} of {triggered.limit} {triggered.label}.{' '}
        <LinkButton href="/settings/billing" variant="link" className="h-auto px-0 text-amber-900 underline dark:text-amber-200">
          Upgrade your plan
        </LinkButton>
      </div>
      <Button variant="ghost-muted" size="icon-xs" onClick={() => setDismissed(true)} aria-label="Dismiss">
        <PlatformIcon icon="IconX" size={14} strokeWidth={2} />
      </Button>
    </div>
  )
}
```

**Step 1.2 — Create `packages/web/modules/billing/components/upgrade-modal.tsx`:**

Triggered programmatically when any fetch returns 402 PLAN_LIMIT. Provides a context provider so any component can call `useUpgradeModal().trigger(info)`.

```tsx
"use client"

import { createContext, useCallback, useContext, useState } from "react"
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter } from "@/modules/common/ui/dialog"
import { Button, LinkButton } from "@/modules/common/ui/button"
import { Card, CardContent } from "@/modules/common/ui/card"

type PlanLimitInfo = { resource: 'units' | 'properties' | 'members'; limit: number; current: number; upgrade_to: 'growth' | 'pro' }
type Ctx = { trigger: (info: PlanLimitInfo) => void }
const UpgradeModalContext = createContext<Ctx | null>(null)

const UPGRADE_LIMITS: Record<'growth' | 'pro', { units: string; properties: string; members: string }> = {
  growth: { units: '20 units', properties: '5 properties', members: '4 team members' },
  pro:    { units: '50 units', properties: 'Unlimited',    members: 'Unlimited' },
}

export function UpgradeModalProvider({ children }: { children: React.ReactNode }) {
  const [info, setInfo] = useState<PlanLimitInfo | null>(null)
  const trigger = useCallback((i: PlanLimitInfo) => setInfo(i), [])

  return (
    <UpgradeModalContext.Provider value={{ trigger }}>
      {children}
      <Dialog open={!!info} onOpenChange={(o) => { if (!o) setInfo(null) }}>
        {info && (
          <DialogContent className="sm:max-w-md">
            <DialogHeader>
              <DialogTitle>You've reached your {info.resource} limit</DialogTitle>
              <DialogDescription>
                Your plan allows {info.limit} {info.resource}. You're using {info.current}. Upgrade to {info.upgrade_to} for more.
              </DialogDescription>
            </DialogHeader>
            <div className="flex gap-3">
              <Card size="sm" className="flex-1">
                <CardContent className="flex flex-col gap-1 text-sm">
                  <span className="font-medium">Current</span>
                  <span>{info.limit} {info.resource}</span>
                </CardContent>
              </Card>
              <Card size="sm" className="flex-1 ring-2 ring-primary">
                <CardContent className="flex flex-col gap-1 text-sm">
                  <span className="font-medium capitalize">{info.upgrade_to}</span>
                  <span>{UPGRADE_LIMITS[info.upgrade_to][info.resource]}</span>
                </CardContent>
              </Card>
            </div>
            <DialogFooter>
              <Button variant="outline" onClick={() => setInfo(null)}>Maybe later</Button>
              <LinkButton href="/settings/billing" variant="default" onClick={() => setInfo(null)}>View billing plans</LinkButton>
            </DialogFooter>
          </DialogContent>
        )}
      </Dialog>
    </UpgradeModalContext.Provider>
  )
}

export function useUpgradeModal() {
  const ctx = useContext(UpgradeModalContext)
  if (!ctx) throw new Error('useUpgradeModal must be inside UpgradeModalProvider')
  return ctx
}
```

CRITICAL: Place under the `(auth)` layout group (NOT `(app)`).
```tsx
"use client"

import { useState, useEffect } from "react"
import { useParams, useRouter } from "next/navigation"
import { authClient } from "@/lib/auth-client"
import { Button, LinkButton } from "@/modules/common/ui/button"
import { Card, CardHeader, CardTitle, CardDescription, CardContent, CardFooter } from "@/modules/common/ui/card"
import { Skeleton } from "@/modules/common/ui/skeleton"
import { toast } from "sonner"

export default function AcceptInvitationPage() {
  const params = useParams<{ id: string }>()
  const router = useRouter()
  const { data: session } = authClient.useSession()
  const [invitation, setInvitation] = useState<{ email: string; role: string; organizationName?: string; inviterEmail?: string } | null>(null)
  const [loading, setLoading] = useState(false)

  useEffect(() => {
    void (async () => {
      const { data } = await authClient.organization.getInvitation({ query: { id: params.id } })
      if (data) setInvitation(data as never)
    })()
  }, [params.id])

  if (!session?.user) {
    return (
      <Card>
        <CardHeader><CardTitle>You're invited to Villuh</CardTitle><CardDescription>Sign in or create an account to accept.</CardDescription></CardHeader>
        <CardFooter className="justify-end gap-2">
          <LinkButton href={`/login?redirect=/accept-invitation/${params.id}`} variant="outline">Sign in</LinkButton>
          <LinkButton href={`/register?redirect=/accept-invitation/${params.id}`} variant="default">Sign up</LinkButton>
        </CardFooter>
      </Card>
    )
  }

  if (!invitation) return <Skeleton className="h-32" />

  return (
    <Card>
      <CardHeader>
        <CardTitle>You've been invited to {invitation.organizationName ?? 'an organization'}</CardTitle>
        <CardDescription>Role: {invitation.role}</CardDescription>
      </CardHeader>
      <CardFooter className="justify-end gap-2">
        <Button
          variant="outline"
          disabled={loading}
          onClick={async () => {
            setLoading(true)
            try { await authClient.organization.rejectInvitation({ invitationId: params.id }); router.push('/') }
            finally { setLoading(false) }
          }}
        >
          Decline
        </Button>
        <Button
          variant="default"
          disabled={loading}
          onClick={async () => {
            setLoading(true)
            try {
              await authClient.organization.acceptInvitation({ invitationId: params.id })
              toast.success("Joined")
              router.push('/')
            } catch {
              toast.error("Could not accept invitation.")
            } finally { setLoading(false) }
          }}
        >
          Accept invitation
        </Button>
      </CardFooter>
    </Card>
  )
}
```

```tsx
}) {
  const [open, setOpen] = useState(false)
  const [loading, setLoading] = useState(false)
  const copy = COPY[kind]

  return (
    <Dialog open={open} onOpenChange={setOpen}>
      <DialogTrigger asChild>{trigger}</DialogTrigger>
      <DialogContent className="sm:max-w-md">
        <DialogHeader>
          <DialogTitle>{copy.title}</DialogTitle>
          <DialogDescription>{copy.body}</DialogDescription>
        </DialogHeader>
        <DialogFooter>
          <Button variant="outline" onClick={() => setOpen(false)}>{copy.cancel}</Button>
          <Button
            variant="destructive"
            disabled={loading}
            onClick={async () => {
              setLoading(true)
              try {
                const path = kind === 'stripe' ? 'stripe-connect' : 'docusign-connect'
                const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/${path}/disconnect`, {
                  method: 'POST', credentials: 'include',
                })
                if (!res.ok) throw new Error('disconnect failed')
                toast.success("Disconnected")
                setOpen(false)
                onDisconnected()
              } catch {
                toast.error("Something went wrong. Please try again.")
              } finally { setLoading(false) }
            }}
          >
            Disconnect
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}
```

```tsx
"use client"

import { useState, useEffect } from "react"
import { IntegrationCard } from "@/modules/settings/components/integration-card"
import { Skeleton } from "@/modules/common/ui/skeleton"
import { toast } from "sonner"
import { authClient } from "@/lib/auth-client"

export default function IntegrationsPage() {
  const [stripeConnected, setStripeConnected] = useState<boolean | null>(null)
  const [docusignConnected, setDocusignConnected] = useState<boolean | null>(null)

  const refetch = async () => {
    const [s, d] = await Promise.all([
      fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/stripe-connect/status`, { credentials: 'include' }),
      fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/docusign-connect/status`, { credentials: 'include' }),
    ])
    if (s.ok) { const j = await s.json() as { connected: boolean }; setStripeConnected(j.connected) }
    if (d.ok) { const j = await d.json() as { connected: boolean }; setDocusignConnected(j.connected) }
  }
  useEffect(() => { void refetch() }, [])

  if (stripeConnected === null || docusignConnected === null) return <Skeleton className="h-32" />

  // Stripe Connect OAuth URL — landlord_id used as state for CSRF (Plan 02-08 verifies state on callback)
  const handleStripeConnect = async () => {
    const { data: session } = await authClient.getSession()
    const orgId = session?.session?.activeOrganizationId
    if (!orgId) { toast.error("Could not start connection"); return }
    const callback = `${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/stripe-connect/callback`
    const url = `https://connect.stripe.com/oauth/authorize?response_type=code&client_id=${process.env.NEXT_PUBLIC_STRIPE_CLIENT_ID}&scope=read_write&redirect_uri=${encodeURIComponent(callback)}&state=${orgId}`
    window.location.href = url
  }

  const handleDocusignConnect = async () => {
    // Phase 2 stub — D-48 says UI ships now, full OAuth in Phase 4
    const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/docusign-connect/initiate`, { credentials: 'include' })
    const json = await res.json() as { url: string | null; message?: string }
    if (json.url) {
      window.location.href = json.url
    } else {
      toast.message(json.message ?? "DocuSign connection lands in Phase 4.")
    }
  }

  return (
    <div className="flex max-w-3xl flex-col gap-6">
      <h1 className="font-heading text-2xl font-medium">Integrations</h1>
      <IntegrationCard
        kind="stripe"
        name="Stripe Connect"
        description="Receive tenant deposit payments through your Stripe account."
        connected={stripeConnected}
        connectCta="Connect Stripe Account"
        onConnect={handleStripeConnect}
        onChanged={refetch}
      />
      <IntegrationCard
        kind="docusign"
        name="DocuSign"
        description="Send leases for e-signature directly from Villuh."
        connected={docusignConnected}
        connectCta="Connect DocuSign"
        onConnect={handleDocusignConnect}
        onChanged={refetch}
      />
    </div>
  )
}
```
