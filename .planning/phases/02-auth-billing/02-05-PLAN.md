```typescript
import { sql } from 'drizzle-orm'

/**
 * Look up the Stripe subscription record for an org.
 * Better Auth Stripe plugin maintains the `subscription` table with reference_id = org.id.
 * Returns null if no subscription exists yet.
 */
export interface OrgSubscription {
  stripe_customer_id: string
  stripe_subscription_id: string | null
  stripe_subscription_item_id: string | null
  status: string
  plan: string | null
}

export async function getOrgSubscription(orgId: string): Promise<OrgSubscription | null> {
  // Use raw SQL because Better Auth subscription table column names depend on plugin generation.
  // Adjust column names if the generated schema differs.
  return withTenant(orgId, 'system', async (tx) => {
    const rows = await tx.execute(sql`
      SELECT
        "stripeCustomerId" AS stripe_customer_id,
        "stripeSubscriptionId" AS stripe_subscription_id,
        "stripeSubscriptionItemId" AS stripe_subscription_item_id,
        status,
        plan
      FROM "subscription"
      WHERE "referenceId" = ${orgId}
      ORDER BY "createdAt" DESC
      LIMIT 1
    `) as unknown as OrgSubscription[]
    return rows[0] ?? null
  })
}
```

If the Better Auth-generated `subscription` table uses snake_case (`stripe_customer_id`, `stripe_subscription_id`, `reference_id`, `created_at`), adjust the column references accordingly. Verify by reading `packages/db/src/schema/auth.ts`.

**Step 1.2 — Create `packages/server/src/routes/billing/trial-start.ts`:**

```typescript
import { Hono } from 'hono'
import { authMiddleware, type AuthVariables } from '../../middleware/auth.js'
import { requireTenant } from '../../middleware/require-tenant.js'
import { requirePermission } from '../../middleware/require-permission.js'
import { stripeClient } from '../../auth.js'
import { getOrCreateStripeCustomer, markSetupCompleted, syncOrgSubscriptionTier } from '../../lib/billing.js'
import { HTTPException } from 'hono/http-exception'

export const trialStart = new Hono<{ Variables: AuthVariables }>()
  .post('/', authMiddleware, requirePermission('admin'), async (c) => {
    const { landlordId } = await requireTenant(c)
    const user = c.get('user')!

    // Idempotency: if org already has an ACTIVE Stripe subscription record, return its state instead of creating a duplicate.
    // Note: org.subscription_status='trial_pending' (default per D-19/D-33/D-39 reconciliation) does NOT have a Stripe subscription row yet —
    // getOrgSubscription returns null for trial_pending users so this branch correctly proceeds to create the trial subscription.
    const { getOrgSubscription } = await import('../../lib/billing.js')
    const existing = await getOrgSubscription(landlordId)
    if (existing && existing.status !== 'canceled' && existing.status !== 'incomplete_expired') {
      return c.json({ ok: true, alreadyActive: true, status: existing.status })
    }

    const stripeCustomerId = await getOrCreateStripeCustomer(stripeClient, landlordId, user.email)

    let subscription
    try {
      subscription = await stripeClient.subscriptions.create({
        customer: stripeCustomerId,
        items: [{ price: process.env.STRIPE_PRO_PRICE_ID!, quantity: 0 }],
        trial_period_days: 30,                                                       // D-19
        trial_settings: { end_behavior: { missing_payment_method: 'cancel' } },     // D-19 (no card required)
        payment_settings: { save_default_payment_method: 'on_subscription' },
        metadata: { landlord_id: landlordId },
      })
    } catch (err) {
      throw new HTTPException(500, { message: `Stripe error: ${(err as Error).message}` })
    }

    // Sync org row immediately: subscription_status='trial_pending' → 'trialing' (D-19/D-33/D-39 reconciliation).
    // Better Auth webhook will fire async; we set state synchronously to avoid race in wizard step 4 navigation.
    await syncOrgSubscriptionTier(landlordId, 'pro', 'trialing')

    // Mark wizard step 4 complete (D-39)
    await markSetupCompleted(landlordId)

    return c.json({
      ok: true,
      subscriptionId: subscription.id,
      trialEnd: subscription.trial_end,
      tier: 'pro',
      status: 'trialing',
    })
  })
```

**Step 1.4 — Create `packages/server/src/routes/billing/subscribe.ts`:**

```typescript
import { Hono } from 'hono'
import { z } from 'zod'
import { zValidator } from '@hono/zod-validator'
import { authMiddleware, type AuthVariables } from '../../middleware/auth.js'
import { requireTenant } from '../../middleware/require-tenant.js'
import { requirePermission } from '../../middleware/require-permission.js'
import { stripeClient } from '../../auth.js'
import { getOrCreateStripeCustomer, getOrgSubscription, syncOrgSubscriptionTier } from '../../lib/billing.js'
import { HTTPException } from 'hono/http-exception'

const subscribeSchema = z.object({
  tier: z.enum(['starter', 'growth', 'pro']),
  paymentMethodId: z.string(),
  unitCount: z.number().int().min(0).default(0),
})

const PRICE_ENV: Record<string, string> = {
  starter: 'STRIPE_STARTER_PRICE_ID',
  growth: 'STRIPE_GROWTH_PRICE_ID',
  pro: 'STRIPE_PRO_PRICE_ID',
}

export const subscribe = new Hono<{ Variables: AuthVariables }>()
  .post('/', authMiddleware, requirePermission('admin'), zValidator('json', subscribeSchema), async (c) => {
    const { landlordId } = await requireTenant(c)
    const user = c.get('user')!
    const { tier, paymentMethodId, unitCount } = c.req.valid('json')
    const priceId = process.env[PRICE_ENV[tier]]
    if (!priceId) throw new HTTPException(500, { message: `Missing env ${PRICE_ENV[tier]}` })

    const stripeCustomerId = await getOrCreateStripeCustomer(stripeClient, landlordId, user.email)

    // Attach the PM to the customer so it can be the default
    await stripeClient.paymentMethods.attach(paymentMethodId, { customer: stripeCustomerId })
    await stripeClient.customers.update(stripeCustomerId, {
      invoice_settings: { default_payment_method: paymentMethodId },
    })

    // If the customer already has a subscription, update its plan instead of creating a 2nd
    const existing = await getOrgSubscription(landlordId)
    if (existing?.stripe_subscription_id) {
      const sub = await stripeClient.subscriptions.update(existing.stripe_subscription_id, {
        items: [{ id: existing.stripe_subscription_item_id!, price: priceId, quantity: unitCount }],
        default_payment_method: paymentMethodId,
        proration_behavior: 'create_prorations',
        trial_end: 'now',                  // exit trial immediately on first paid subscribe
      })
      await syncOrgSubscriptionTier(landlordId, tier, sub.status)
      return c.json({ ok: true, subscriptionId: sub.id, tier, status: sub.status })
    }

    const sub = await stripeClient.subscriptions.create({
      customer: stripeCustomerId,
      items: [{ price: priceId, quantity: unitCount }],
      default_payment_method: paymentMethodId,
      metadata: { landlord_id: landlordId },
    })
    await syncOrgSubscriptionTier(landlordId, tier, sub.status)
    return c.json({ ok: true, subscriptionId: sub.id, tier, status: sub.status })
  })
```

```typescript
import { authMiddleware, type AuthVariables } from '../../middleware/auth.js'
import { requireTenant } from '../../middleware/require-tenant.js'
import { requirePermission } from '../../middleware/require-permission.js'
import { stripeClient } from '../../auth.js'
import { getOrgSubscription, syncOrgSubscriptionTier } from '../../lib/billing.js'
import { HTTPException } from 'hono/http-exception'

const upgradeSchema = z.object({ tier: z.enum(['starter', 'growth', 'pro']) })
const PRICE_ENV: Record<string, string> = {
  starter: 'STRIPE_STARTER_PRICE_ID',
  growth: 'STRIPE_GROWTH_PRICE_ID',
  pro: 'STRIPE_PRO_PRICE_ID',
```

```typescript
import { authMiddleware, type AuthVariables } from '../../middleware/auth.js'
import { requireTenant } from '../../middleware/require-tenant.js'
import { requirePermission } from '../../middleware/require-permission.js'
import { stripeClient } from '../../auth.js'
import { getOrgSubscription } from '../../lib/billing.js'
import { z } from 'zod'
import { zValidator } from '@hono/zod-validator'

const updateSchema = z.object({ paymentMethodId: z.string() })

export const paymentMethod = new Hono<{ Variables: AuthVariables }>()
  .get('/', authMiddleware, requirePermission('admin'), async (c) => {
    const { landlordId } = await requireTenant(c)
    const sub = await getOrgSubscription(landlordId)
    if (!sub?.stripe_customer_id) return c.json({ paymentMethod: null })

    const customer = await stripeClient.customers.retrieve(sub.stripe_customer_id, {
      expand: ['invoice_settings.default_payment_method'],
    })
    if ('deleted' in customer && customer.deleted) return c.json({ paymentMethod: null })
    const pm = (customer as import('stripe').Stripe.Customer).invoice_settings?.default_payment_method
    if (!pm || typeof pm === 'string') return c.json({ paymentMethod: null })
    return c.json({
      paymentMethod: {
        id: pm.id,
        brand: pm.card?.brand,
        last4: pm.card?.last4,
        exp_month: pm.card?.exp_month,
        exp_year: pm.card?.exp_year,
      },
    })
  })
  .put('/', authMiddleware, requirePermission('admin'), zValidator('json', updateSchema), async (c) => {
    const { landlordId } = await requireTenant(c)
    const { paymentMethodId } = c.req.valid('json')
    const sub = await getOrgSubscription(landlordId)
    if (!sub?.stripe_customer_id) return c.json({ error: 'No customer' }, 400)
    await stripeClient.paymentMethods.attach(paymentMethodId, { customer: sub.stripe_customer_id })
    await stripeClient.customers.update(sub.stripe_customer_id, {
      invoice_settings: { default_payment_method: paymentMethodId },
    })
    return c.json({ ok: true })
  })
```
