**Step 1.1 — Create `packages/web/modules/billing/hooks/use-billing-status.ts`:**

Reactive hook that fetches `/api/me` and exposes the D-44 shape. Auto-refetch on login/logout via `authClient.useSession`. Plan 02-12's UsageWarningBanner consumes the same hook.

```typescript
"use client"

import { useEffect, useState, useCallback } from "react"
import { authClient } from "@/lib/auth-client"

export type BillingStatus = {
  user: { id: string; name: string; email: string; image: string | null }
  org: { id: string; name: string; slug: string; subscription_tier: string; subscription_status: string; setup_completed: boolean }
  tier: string
  unit_count: number; unit_limit: number
  property_count: number; property_limit: number
  member_count: number; member_limit: number
  subscription_status: string
  permissions: { write: boolean; financial: boolean; communications: boolean; admin: boolean }
}

export function useBillingStatus() {
  const { data: session } = authClient.useSession()
  const [status, setStatus] = useState<BillingStatus | null>(null)
  const [error, setError] = useState<Error | null>(null)

  const refetch = useCallback(async () => {
    try {
      const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/me`, { credentials: 'include' })
      if (!res.ok) throw new Error(`me: ${res.status}`)
      setStatus(await res.json() as BillingStatus)
    } catch (e) { setError(e as Error) }
  }, [])

  useEffect(() => { if (session?.user) void refetch() }, [session?.user, refetch])

  return { status, error, refetch }
}
```

**Step 1.2 — Create `packages/web/modules/billing/hooks/use-stripe-elements.ts`:**

Helper to load Stripe.js once and fetch a SetupIntent client secret on demand.

```typescript
"use client"

import { useEffect, useState } from "react"
import { loadStripe, type Stripe } from "@stripe/stripe-js"

let stripePromise: Promise<Stripe | null> | null = null
function getStripe() {
  if (!stripePromise) {
    stripePromise = loadStripe(process.env.NEXT_PUBLIC_STRIPE_PUBLISHABLE_KEY!)
  }
  return stripePromise
}

export function useSetupIntent(active: boolean) {
  const [clientSecret, setClientSecret] = useState<string | null>(null)
  const [stripeInstance, setStripeInstance] = useState<Stripe | null>(null)

  useEffect(() => {
    if (!active) return
    void (async () => {
      const [stripe, csRes] = await Promise.all([
        getStripe(),
        fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/setup-intent`, {
          method: 'POST', credentials: 'include',
        }),
      ])
      const cs = await csRes.json() as { clientSecret: string }
      setStripeInstance(stripe)
      setClientSecret(cs.clientSecret)
    })()
  }, [active])

  return { clientSecret, stripeInstance }
}
```

**Step 2.1 — Create `packages/web/modules/billing/components/payment-method-form.tsx`:**

Stripe PaymentElement embedded (no Checkout redirect, per D-27). Used during the plan-upgrade flow. Uses `<Elements>` provider + `useStripe`/`useElements` hooks. On submit: stripe.confirmSetup → POST /api/billing/subscribe with paymentMethodId.

```tsx
"use client"

import { Elements, PaymentElement, useStripe, useElements } from "@stripe/react-stripe-js"
import type { Stripe, StripeElementsOptions } from "@stripe/stripe-js"
import { Button } from "@/modules/common/ui/button"
import { Card, CardContent } from "@/modules/common/ui/card"
import { useState } from "react"
import { toast } from "sonner"
import { useSetupIntent } from "../hooks/use-stripe-elements"

function PaymentForm({ tier, unitCount, onComplete }: { tier: 'starter' | 'growth' | 'pro'; unitCount: number; onComplete: () => void }) {
  const stripe = useStripe()
  const elements = useElements()
  const [loading, setLoading] = useState(false)

  return (
    <form
      className="flex flex-col gap-4"
      onSubmit={async (e) => {
        e.preventDefault()
        if (!stripe || !elements) return
        setLoading(true)
        try {
          const { error: confirmError, setupIntent } = await stripe.confirmSetup({
            elements,
            redirect: 'if_required',
            confirmParams: { return_url: `${window.location.origin}/settings/billing` },
          })
          if (confirmError) {
            toast.error(confirmError.message ?? "Payment method could not be saved.")
            return
          }
          const paymentMethodId = setupIntent?.payment_method as string
          if (!paymentMethodId) {
            toast.error("Could not save payment method.")
            return
          }
          const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/subscribe`, {
            method: 'POST', credentials: 'include',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({ tier, paymentMethodId, unitCount }),
          })
          if (!res.ok) {
            toast.error("Could not start subscription.")
            return
          }
          toast.success("Plan updated")
          onComplete()
        } finally { setLoading(false) }
      }}
    >
      <PaymentElement />
      <Button type="submit" size="lg" disabled={loading || !stripe || !elements}>
        Save payment method &amp; upgrade
      </Button>
    </form>
  )
}

export function PaymentMethodForm({ tier, unitCount, onComplete }: { tier: 'starter' | 'growth' | 'pro'; unitCount: number; onComplete: () => void }) {
  const { clientSecret, stripeInstance } = useSetupIntent(true)
  if (!clientSecret || !stripeInstance) {
    return <div className="text-sm text-muted-foreground">Loading payment form…</div>
  }
  const options: StripeElementsOptions = { clientSecret, appearance: { theme: 'stripe' } }
  return (
    <Card>
      <CardContent>
        <Elements stripe={stripeInstance} options={options}>
          <PaymentForm tier={tier} unitCount={unitCount} onComplete={onComplete} />
        </Elements>
      </CardContent>
    </Card>
  )
}
```

   - `stripe.confirmSetup({ elements, redirect: 'if_required', confirmParams: { return_url } })` to tokenize the new card.
   - `PUT /api/billing/payment-method` with `{ paymentMethodId }` to persist replacement on Stripe customer.
   - Refetch the GET to display the new card.

This is distinct from the plan-upgrade flow (`PaymentMethodForm`) which calls `/api/billing/subscribe`. Updating a saved card while the plan does not change must NOT touch the subscription.

```tsx
"use client"

import { Elements, PaymentElement, useStripe, useElements } from "@stripe/react-stripe-js"
import type { StripeElementsOptions } from "@stripe/stripe-js"
import { Button } from "@/modules/common/ui/button"
import { Card, CardHeader, CardTitle, CardDescription, CardContent, CardFooter } from "@/modules/common/ui/card"
import { Skeleton } from "@/modules/common/ui/skeleton"
import { useEffect, useState, useCallback } from "react"
import { toast } from "sonner"
import { useSetupIntent } from "../hooks/use-stripe-elements"

type SavedPaymentMethod = {
  brand: string          // 'visa' | 'mastercard' | 'amex' | etc.
  last4: string          // '4242'
  exp_month: number      // 1-12
  exp_year: number       // 4-digit year
}

function UpdateCardForm({ onComplete }: { onComplete: () => void }) {
  const stripe = useStripe()
  const elements = useElements()
  const [loading, setLoading] = useState(false)

  return (
    <form
      className="flex flex-col gap-4"
      onSubmit={async (e) => {
        e.preventDefault()
        if (!stripe || !elements) return
        setLoading(true)
        try {
          const { error: confirmError, setupIntent } = await stripe.confirmSetup({
            elements,
            redirect: 'if_required',
            confirmParams: { return_url: `${window.location.origin}/settings/billing` },
          })
          if (confirmError) {
            toast.error(confirmError.message ?? "Card could not be saved.")
            return
          }
          const paymentMethodId = setupIntent?.payment_method as string
          if (!paymentMethodId) {
            toast.error("Could not save card.")
            return
          }
          // PUT to /api/billing/payment-method (NOT subscribe — plan does not change here)
          const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/payment-method`, {
            method: 'PUT', credentials: 'include',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({ paymentMethodId }),
          })
          if (!res.ok) {
            toast.error("Could not update card.")
            return
          }
          toast.success("Card updated")
          onComplete()
        } finally { setLoading(false) }
      }}
    >
      <PaymentElement />
      <Button type="submit" size="lg" disabled={loading || !stripe || !elements}>
        Save card
      </Button>
    </form>
  )
}

export function PaymentMethodCard() {
  const [pm, setPm] = useState<SavedPaymentMethod | null | undefined>(undefined)  // undefined = loading, null = none
  const [editing, setEditing] = useState(false)
  const { clientSecret, stripeInstance } = useSetupIntent(editing)

  const refetch = useCallback(async () => {
    try {
      const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:3001'}/api/billing/payment-method`, { credentials: 'include' })
      if (!res.ok) { setPm(null); return }
      const json = await res.json() as { paymentMethod: SavedPaymentMethod | null }
      setPm(json.paymentMethod)
    } catch { setPm(null) }
  }, [])

  useEffect(() => { void refetch() }, [refetch])

  const brandLabel = (b: string) => b.charAt(0).toUpperCase() + b.slice(1)

  return (
    <Card>
      <CardHeader>
        <CardTitle>Payment method</CardTitle>
        <CardDescription>
          {pm === undefined
            ? "Loading…"
            : pm
            ? "Used for recurring subscription charges and overages."
            : "Add a card to continue after your trial ends."}
        </CardDescription>
      </CardHeader>
      <CardContent className="flex flex-col gap-3">
        {pm === undefined && <Skeleton className="h-6 w-48" />}
        {pm === null && (
          <p className="text-sm text-muted-foreground">No payment method saved</p>
        )}
        {pm && (
          <p className="text-sm">
            <span className="font-medium">{brandLabel(pm.brand)}</span>
            {' '}····{pm.last4}
            <span className="text-muted-foreground"> · Exp {String(pm.exp_month).padStart(2, '0')}/{pm.exp_year}</span>
          </p>
        )}
        {editing && clientSecret && stripeInstance && (
          <div className="mt-2">
            <Elements
              stripe={stripeInstance}
              options={{ clientSecret, appearance: { theme: 'stripe' } } satisfies StripeElementsOptions}
            >
              <UpdateCardForm onComplete={() => { setEditing(false); void refetch() }} />
            </Elements>
          </div>
        )}
        {editing && (!clientSecret || !stripeInstance) && (
          <p className="text-sm text-muted-foreground">Loading payment form…</p>
        )}
      </CardContent>
      <CardFooter className="justify-end">
        {!editing && (
          <Button variant="outline" onClick={() => setEditing(true)}>
            {pm ? "Update card" : "Add card"}
          </Button>
        )}
        {editing && (
          <Button variant="ghost-muted" onClick={() => setEditing(false)}>Cancel</Button>
        )}
      </CardFooter>
    </Card>
  )
}
```
