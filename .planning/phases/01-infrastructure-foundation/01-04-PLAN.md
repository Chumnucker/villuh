```sql
-- 1. Create audit_log table with monthly range partitioning
CREATE TABLE audit_log (
  id          uuid NOT NULL DEFAULT gen_random_uuid(),
  landlord_id uuid NOT NULL,
  table_name  text NOT NULL,
  row_id      uuid NOT NULL,
  operation   text NOT NULL CHECK (operation IN ('INSERT', 'UPDATE', 'DELETE')),
  old_values  jsonb,
  new_values  jsonb,
  actor_id    text NOT NULL DEFAULT 'system',
  created_at  timestamptz NOT NULL DEFAULT now()
) PARTITION BY RANGE (created_at);

-- 2. Pre-create partitions for current month and next two months
--    CRITICAL: Postgres WILL FAIL on INSERT if no matching partition exists.
--    A future phase adds a monthly cron job to pre-create partitions.
--    Add partitions here manually if Phase 1 is being run after May 2026.
CREATE TABLE audit_log_y2026m04 PARTITION OF audit_log
  FOR VALUES FROM ('2026-04-01') TO ('2026-05-01');
CREATE TABLE audit_log_y2026m05 PARTITION OF audit_log
  FOR VALUES FROM ('2026-05-01') TO ('2026-06-01');
CREATE TABLE audit_log_y2026m06 PARTITION OF audit_log
  FOR VALUES FROM ('2026-06-01') TO ('2026-07-01');
CREATE TABLE audit_log_y2026m07 PARTITION OF audit_log
  FOR VALUES FROM ('2026-07-01') TO ('2026-08-01');

-- 3. Enable RLS on audit_log (landlords can only read their own audit rows)
ALTER TABLE audit_log ENABLE ROW LEVEL SECURITY;
CREATE POLICY audit_log_tenant_isolation ON audit_log
  AS PERMISSIVE FOR SELECT
  TO villuh_app
  USING (landlord_id = current_setting('app.current_landlord_id', TRUE)::uuid);

-- 4. Trigger function: writes an audit row on every INSERT/UPDATE/DELETE
CREATE OR REPLACE FUNCTION audit_log_trigger_fn()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO audit_log (landlord_id, table_name, row_id, operation, old_values, new_values, actor_id)
  VALUES (
    COALESCE(
      (CASE TG_OP WHEN 'DELETE' THEN OLD.landlord_id ELSE NEW.landlord_id END),
      '00000000-0000-0000-0000-000000000000'::uuid
    ),
    TG_TABLE_NAME,
    COALESCE(
      (CASE TG_OP WHEN 'DELETE' THEN OLD.id ELSE NEW.id END)
    ),
    TG_OP,
    CASE TG_OP WHEN 'INSERT' THEN NULL ELSE to_jsonb(OLD) END,
    CASE TG_OP WHEN 'DELETE' THEN NULL ELSE to_jsonb(NEW) END,
    COALESCE(current_setting('app.current_user_id', TRUE), 'system')
  );
  RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 5. Immutability trigger: prevent UPDATE or DELETE on audit_log rows
CREATE OR REPLACE FUNCTION audit_log_immutable_fn()
RETURNS TRIGGER AS $$
BEGIN
  RAISE EXCEPTION 'audit_log rows are immutable — modification is forbidden';
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER audit_log_immutable
  BEFORE UPDATE OR DELETE ON audit_log
  FOR EACH ROW EXECUTE FUNCTION audit_log_immutable_fn();

-- 6. Attach audit trigger to the test_tenant_data table (demo of the pattern)
--    All future tenant tables will get this same trigger in their respective migrations.
CREATE TRIGGER test_tenant_data_audit
  AFTER INSERT OR UPDATE OR DELETE ON test_tenant_data
  FOR EACH ROW EXECUTE FUNCTION audit_log_trigger_fn();
```

|----------|-------------|
| migration runner → Postgres | Owner-role DDL execution — must never leak to runtime path |
| audit trigger → audit_log | SECURITY DEFINER function writes as owner even when called from villuh_app session |
| audit_log rows | Immutability enforced at DB layer by trigger — application layer cannot bypass |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-01 | Tampering | audit_log immutability | mitigate | audit_log_immutable trigger raises exception on UPDATE/DELETE; tested in Plan 05 integration test |
| T-04-02 | Information Disclosure | audit_log RLS | mitigate | RLS on audit_log uses same landlord_id pattern as other tables; villuh_app can only see own tenant's audit rows |
| T-04-03 | Elevation of Privilege | SECURITY DEFINER trigger | accept | audit_log_trigger_fn runs as owner to write to audit_log, bypassing app-level restrictions — this is intentional; SECURITY DEFINER is required for the audit log to work when called from villuh_app context |
| T-04-04 | Information Disclosure | Missing partition | mitigate | 4 months of partitions pre-created (Apr–Jul 2026); operational gap documented in SQL comment; partition failure surfaces as migration error, not silent data loss |
</threat_model>
