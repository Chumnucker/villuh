```typescript
import { drizzle } from 'drizzle-orm/postgres-js'
import postgres from 'postgres'
import * as schema from '../schema/index.js'
import { sql } from 'drizzle-orm'

// OWNER CONNECTION — use for seeding test data and reading audit rows (bypasses RLS)
// NEVER use this connection for RLS assertions — owner bypasses RLS silently
export function createOwnerClient() {
  if (!process.env.DATABASE_URL) {
    throw new Error('DATABASE_URL is required for test setup (owner role for seeding)')
  }
  const client = postgres(process.env.DATABASE_URL, { max: 1 })
  return {
    db: drizzle({ client, schema }),
    end: () => client.end(),
  }
}

// RESTRICTED CONNECTION — use for RLS assertions (connects as villuh_app)
// This is the connection that RLS policies are enforced on
export function createAppClient() {
  if (!process.env.DATABASE_TEST_URL) {
    throw new Error('DATABASE_TEST_URL is required for RLS tests (villuh_app role)')
  }
  const client = postgres(process.env.DATABASE_TEST_URL, { max: 5 })
  return {
    db: drizzle({ client, schema }),
    client,
    end: () => client.end(),
  }
}

// Helper: set tenant context within a transaction (replicates withTenant from @villuh/server)
// Defined here to avoid cross-package dependency in the test
export async function withTestTenant<T>(
  appDb: ReturnType<typeof createAppClient>['db'],
  landlordId: string,
  actorId: string = 'system',
  fn: (tx: Parameters<Parameters<ReturnType<typeof createAppClient>['db']['transaction']>[0]>[0]) => Promise<T>
): Promise<T> {
  return appDb.transaction(async (tx) => {
    await tx.execute(sql`SELECT set_config('app.current_landlord_id', ${landlordId}, TRUE)`)
    await tx.execute(sql`SELECT set_config('app.current_user_id', ${actorId}, TRUE)`)
    return fn(tx)
  })
}
```

**Step 1: Create packages/db/src/__tests__/rls.test.ts**:

```typescript
import { afterAll, beforeAll, describe, expect, it } from 'vitest'
import { eq, sql } from 'drizzle-orm'
import { testTenantData } from '../schema/landlords.js'
import { createAppClient, createOwnerClient, withTestTenant } from './setup.js'

// Test UUIDs — deterministic for easy cleanup
const LANDLORD_A = 'aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa'
const LANDLORD_B = 'bbbbbbbb-bbbb-bbbb-bbbb-bbbbbbbbbbbb'
const ROW_A_ID = 'a0000001-aaaa-aaaa-aaaa-aaaaaaaaaaaa'
const ROW_B_ID = 'b0000001-bbbb-bbbb-bbbb-bbbbbbbbbbbb'

const owner = createOwnerClient()
const app = createAppClient()

beforeAll(async () => {
  // Seed test data using OWNER connection (bypasses RLS for setup)
  // Insert row for landlord A
  await owner.db
    .insert(testTenantData)
    .values({ id: ROW_A_ID, landlordId: LANDLORD_A, data: 'landlord-a-data' })
    .onConflictDoNothing()

  // Insert row for landlord B
  await owner.db
    .insert(testTenantData)
    .values({ id: ROW_B_ID, landlordId: LANDLORD_B, data: 'landlord-b-data' })
    .onConflictDoNothing()
})

afterAll(async () => {
  // Clean up test data using OWNER connection
  await owner.db
    .delete(testTenantData)
    .where(eq(testTenantData.landlordId, LANDLORD_A))
  await owner.db
    .delete(testTenantData)
    .where(eq(testTenantData.landlordId, LANDLORD_B))

  await owner.end()
  await app.end()
})

describe('RLS: cross-tenant isolation', () => {
  it('landlord A cannot see landlord B rows', async () => {
    const results = await withTestTenant(app.db, LANDLORD_A, 'system', async (tx) => {
      return tx.select().from(testTenantData)
    })

    // Landlord A should see only their own row — not landlord B's
    const hasLandlordBData = results.some((row) => row.landlordId === LANDLORD_B)
    expect(hasLandlordBData).toBe(false)
  })

  it('landlord A sees only their own rows', async () => {
    const results = await withTestTenant(app.db, LANDLORD_A, 'system', async (tx) => {
      return tx.select().from(testTenantData)
    })

    expect(results.length).toBeGreaterThan(0)
    expect(results.every((row) => row.landlordId === LANDLORD_A)).toBe(true)
  })

  it('landlord B sees only their own rows', async () => {
    const results = await withTestTenant(app.db, LANDLORD_B, 'system', async (tx) => {
      return tx.select().from(testTenantData)
    })

    expect(results.length).toBeGreaterThan(0)
    expect(results.every((row) => row.landlordId === LANDLORD_B)).toBe(true)
  })

  it('no context set = zero rows visible (RLS blocks everything)', async () => {
    // Query without SET LOCAL — no tenant context = empty result set
    const results = await app.db.execute(
      sql`SELECT * FROM test_tenant_data`
    )
    expect(results.length).toBe(0)
  })
})

describe('Audit log: write and immutability', () => {
  const TEST_ACTOR = 'test-user-integration-123'
  const AUDIT_TEST_LANDLORD = 'cccccccc-cccc-cccc-cccc-cccccccccccc'
  const AUDIT_TEST_ROW_ID = 'c0000001-cccc-cccc-cccc-cccccccccccc'

  it('writes an audit row on INSERT to test_tenant_data', async () => {
    // Insert using the withTestTenant pattern (villuh_app + SET LOCAL)
    await withTestTenant(app.db, AUDIT_TEST_LANDLORD, TEST_ACTOR, async (tx) => {
      await tx
        .insert(testTenantData)
        .values({
          id: AUDIT_TEST_ROW_ID,
          landlordId: AUDIT_TEST_LANDLORD,
          data: 'audit-test-data',
        })
    })

    // Read audit rows as OWNER (owner bypasses RLS — needed to see audit rows without context)
    const auditRows = await owner.db.execute(
      sql`SELECT * FROM audit_log WHERE row_id = ${AUDIT_TEST_ROW_ID}::uuid`
    )

    expect(auditRows.length).toBe(1)
    const auditRow = auditRows[0] as Record<string, unknown>
    expect(auditRow['table_name']).toBe('test_tenant_data')
    expect(auditRow['operation']).toBe('INSERT')
    expect(auditRow['actor_id']).toBe(TEST_ACTOR)
    expect(auditRow['landlord_id']).toBe(AUDIT_TEST_LANDLORD)

    // Clean up
    await owner.db
      .delete(testTenantData)
      .where(eq(testTenantData.id, AUDIT_TEST_ROW_ID))
  })

  it('audit_log rows are immutable — UPDATE raises an exception', async () => {
    // First get an audit row ID to attempt UPDATE on
    const auditRows = await owner.db.execute(
      sql`SELECT id FROM audit_log LIMIT 1`
    )

    if (auditRows.length === 0) {
      // No audit rows yet — skip this test but warn
      console.warn('No audit rows found — run insert test first or ensure seeding creates audit entries')
      return
    }

    const auditId = (auditRows[0] as Record<string, unknown>)['id']

    // Attempt UPDATE — this MUST throw
    await expect(
      owner.db.execute(
        sql`UPDATE audit_log SET actor_id = 'tampered' WHERE id = ${auditId}::uuid`
      )
    ).rejects.toThrow(/immutable/)
  })
})
```
